        \begin{lstlisting}[language=sparql, caption={Replace \texttt{rr:class} definitions with predicate object maps},
        label={lst:sparql_class_expand}]
DELETE {
  ?sm rr:class ?sm_class . }
INSERT {
  ?tm rr:predicateObjectMap [
      rr:predicateMap [
          rr:constant rdf:type ;
          rr:termType rr:IRI ;
      ] ;
      rr:objectMap [
         rr:termType rr:IRI ;
         rr:constant ?sm_class ;
      ] ;
  ] . }
WHERE {
  ?tm rr:subjectMap ?sm . 
  ?sm rr:class ?sm_class . }
\end{lstlisting}

        \centering
\begin{lstlisting}[language=sparql, caption={Replace complex predicate-object maps with
simple predicate-object maps},
label={lst:sparql_simple_pom}]
DELETE{
  ?tm rr:predicateObjectMap  ?pom. 
  ?pom rr:predicateMap ?pm. 
  ?pom rr:objectMap ?om. 
  ?pom rr:graphMap ?gm. }
INSERT{
  ?tm rr:predicateObjectMap [
      rr:predicateMap ?pm;
      rr:objectMap ?om; 
      rr:graphMap ?gm;
  ]. }
WHERE{
  ?tm rr:predicateObjectMap  ?pom. 
  ?pom rr:predicateMap ?pm. 
  ?pom rr:objectMap ?om. 
  OPTIONAL {?pom rr:graphMap ?gm.} } 
\end{lstlisting}

    \begin{minipage}{\textwidth}
\begin{lstlisting}[language=sparql, caption={SPARQL update query to expand shortcuts
for constant-term maps},
label={lst:sparql_shortcut}]
DELETE {
  ?tm rr:subject ?sm_constant .
  ?pompm rr:predicate ?pm_constant .
  ?pomom rr:object ?om_constant .
  ?termMap rr:graph ?gm_constant . }
INSERT {
  ?tm rr:subjectMap [ rr:constant ?sm_constant; rr:termType rr:IRI; ].
  ?pompm rr:predicateMap [ rr:constant ?pm_constant; rr:termType rr:IRI; ].
  ?pomom rr:objectMap [ rr:constant ?om_constant; rr:termType ?otermtype; ].
  ?termMap rr:graphMap [ rr:constant ?gm_constant; rr:termType rr:IRI; ]. }
WHERE {
  {?tm rr:subject ?sm_constant .} UNION
  {?pompm rr:predicate ?om_constant .} UNION
  {?pomom rr:object ?om_constant .
  BIND( IF(isIRI( ?om_constant ), rr:IRI, rr:Literal) AS ?otermtype)
  } UNION
  {?termMap rr:graph ?gm_constant .} }
\end{lstlisting}
    \end{minipage}

    \begin{minipage}[t]{0.4\textwidth}
\begin{lstlisting}[language=sparql, caption={Replace referencing object maps without 
join conditions with simple object map},
label={lst:sparql_selfjoin}]
DELETE{
?om rr:parentTriplesMap ?ptm . }
INSERT{
?om rr:reference  ?ref .
?om rr:template ?template . 
?om rr:constant ?const .
?om rr:termType rr:IRI . }
WHERE{
?om rr:parentTriplesMap ?ptm .
?ptm rr:subjectMap ?sm . 
OPTIONAL{?sm rr:reference ?ref .}
OPTIONAL{?sm rr:template ?template .}
OPTIONAL{?sm rr:constant ?const .}
FILTER NOT EXISTS 
{?om rr:joinCondition ?jc .} } 
\end{lstlisting}
