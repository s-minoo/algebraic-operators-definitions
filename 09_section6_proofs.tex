
\section{Proofs for Algebraic Equivalences}

\subsection{Projection splitting}
\label{proof:proj_splitting}
\begin{proof}
    Let $(A_1, \symMappingInst_1) = \fctProjectOp{\symProjSet}{r}$ and $(A_2, \symMappingInst_2) = \fctProjectOpBig{\symProjSet}{\fctProjectOp{\symProjSet\cup\symProjSet'\!}{r}}$, 
    where $\symProjSet'\! \subseteq \symAttrSubSet$. 
    To show that $(A_1, \symMappingInst_1) = (A_2, \symMappingInst_2)$, we first note that, by Definition~\ref{def:projection_operator}, $A_1 = A_2 = \symProjSet$.
    Hence, it remains to show that $\symMappingInst_1 = \symMappingInst_2$. 

    To show that $\symMappingInst_1 \subseteq \symMappingInst_2$ we consider an arbitrary 
    mapping tuple $\mappingTuple \in \symMappingInst_1$ and show that $\mappingTuple \in \symMappingInst_2$. 
    Given $\mappingTuple \in \symMappingInst_1$ with
    Definition~\ref{def:projection_operator}, it holds that there exists
    mapping tuple $\mappingTuple' \in r$ such that $\mappingTuple$ is the 
    restriction of $\mappingTuple'$  to $\symProjSet$; i.e., $\mappingTuple = \mappingTuple'[\symProjSet]$ and 
    $\mappingTuple(a) = \mappingTuple'(a)$ for all $a \in \symProjSet$. 
    Considering $\symProjSet \cup \symProjSet' \subseteq A$, let $\mappingTuple''\! =
    \mappingTuple'[\symProjSet \cup \symProjSet']$, which, by
    Definition~\ref{def:projection_operator}, also means that
    $\fctDom{\mappingTuple''} = \symProjSet \cup \symProjSet'$ and $\mappingTuple''\! \in
    \fctProjectOp{\symProjSet \cup \symProjSet'\!}{r}$.
    Thus, there exists a mapping tuple 
    $\mappingTuple'''\! \in \fctProjectOpBig{\symProjSet}{ \fctProjectOp{\symProjSet\cup\symProjSet'\!}{r} }$
    such that i)~$\fctDom{\mappingTuple'''} = \symProjSet$ and ii)~$\mappingTuple'''(a) = \mappingTuple''(a)$ for all 
    $a \in \symProjSet$ according to Definition~\ref{def:projection_operator}.
    Now that we have $\fctDom{\mappingTuple'''} = \symProjSet = \fctDom{\mappingTuple}$, and 
    $\mappingTuple'''(a) =  \mappingTuple''(a)= \mappingTuple'(a)= \mappingTuple(a)$ for all $a \in \symProjSet$, 
    we can conclude that $\mappingTuple'''\! =  \mappingTuple$ and, thus,
    $\mappingTuple \in \fctProjectOpBig{\symProjSet}{\fctProjectOp{\symProjSet\cup\symProjSet'}{r}}$; i.e., $\mappingTuple  \in \symMappingInst_2$.

    Lastly, we show that $\symMappingInst_2 \subseteq \symMappingInst_1$ by 
    considering an arbitrary mapping tuple $\mappingTuple \in \symMappingInst_2$ and prove that $\mappingTuple \in \symMappingInst_1$. 
    Given $\mappingTuple \in \symMappingInst_2$ with Definition~\ref{def:projection_operator}, it holds 
    that i)~$\fctDom{\mappingTuple} =  \symProjSet$, and ii)~there exists a mapping
    tuple $\mappingTuple'\! \in \fctProjectOpBig{\symProjSet \cup \symProjSet'\!}{r}$ such 
    that $\mappingTuple(a) = \mappingTuple'(a)$ for all $a \in \symProjSet$.
    Since $\mappingTuple'\! \in \fctProjectOpBig{\symProjSet \cup \symProjSet'\!}{r}$, 
    it holds that i)~$\fctDom{\mappingTuple'} = \symProjSet \cup \symProjSet'\!$ and 
    ii)~there exists a mapping tuple $\mappingTuple''\! \in r$ such that 
    $\mappingTuple'(a) = \mappingTuple''(a)$ for all $a \in \symProjSet \cup \symProjSet'\!$. 
    Considering $\symProjSet \subseteq \symAttrSubSet$, let 
    $\mappingTuple'''\! = \mappingTuple''[\symProjSet]$, which, by Definition~\ref{def:projection_operator}, 
    $\fctDom{\mappingTuple'''} = \symProjSet$ and $\mappingTuple'''\! \in \fctProjectOp{\symProjSet}{r}$. 
    Combining everything, we have i)~$\fctDom{\mappingTuple'''} = \symProjSet = \fctDom{\mappingTuple}$ and 
    ii)~$\mappingTuple'''(a) = \mappingTuple''(a) = \mappingTuple'(a) = \mappingTuple(a)$ for all $a \in \symProjSet$. 
    Consequently, $\mappingTuple'''\! = \mappingTuple$ and $\mappingTuple \in \fctProjectOpDflt{r}$; 
    i.e., $\mappingTuple \in \symMappingInst_1$.
    \qed
\end{proof}



\subsection{Projection into Source Operator}%
\label{proof:projection_into_source_operator}

\begin{proof}
    Let $(\symAttrSubSet,\symMappingInst) = \fctProjectOpBig{\symProjSet}{ \sourceOpDflt }$, 
    and $(\symAttrSubSet',\symMappingInst') = \sourceOp{\dataSource}{\queryExpr}{\symAttrQueryMap'}$, 
    where $\symProjSet = \fctDom{\symAttrQueryMap'}$.
    To show that $(\symAttrSubSet,\symMappingInst) = (\symAttrSubSet',\symMappingInst')$, 
    we notice that, by Definition~\ref{def:projection_operator}, $\symAttrSubSet = \symProjSet$, 
    and by Definition~\ref{def:source_operator}, $\symAttrSubSet' = \fctDom{\symAttrQueryMap'} = \symProjSet$. 
    Hence, we have $\symAttrSubSet = \symAttrSubSet'$ and, thus, it remains to
    show that $\symMappingInst = \symMappingInst'$.

    We first show that $\symMappingInst \subseteq \symMappingInst'$ by considering
    an arbitrary mapping tuple $\mappingTuple \in \symMappingInst$ and show 
    that $\mappingTuple \in \symMappingInst'$.
    Given $\mappingTuple \in \symMappingInst$ with Definition~\ref{def:projection_operator}, 
    it holds that there exists a mapping tuple $\mappingTuple'_{\dataObject} \in \sourceOpDflt$ such 
    that $\mappingTuple$ is the restriction of $\mappingTuple'_{\dataObject}$ to $\symProjSet$;
    i.e., $\mappingTuple = \mappingTuple'_{\dataObject}[\symProjSet]$, 
	$\fctDom{\mappingTuple} = \symProjSet$, and 
	$\mappingTuple(\attr) = \mappingTuple'_{\dataObject}(\attr)$ for all $\attr \in \symProjSet$. 
	Since $\mappingTuple'_{\dataObject} \in \sourceOpDflt$ with Definition~\ref{def:source_operator}, it holds that
	i)~$\dataObject \in \fctEval{\bigDataObject}{\queryExpr}$,
	ii)~$X_{\dataObject} = \bigtimes_{\attr \in \fctDom{\symAttrQueryMap}}
		\big\{ (\attr, \var) \ \big| \
		\var \in \fctCbeval{\bigDataObject}{\dataObject}{\queryExpr'} \text{ with }  \queryExpr'\! = \symAttrQueryMap(\attr) \big\}$, 
	iii)~$\fctDom{\mappingTuple'_{\dataObject}} = \fctDom{\symAttrQueryMap}$, and
	iv)~$\mappingTuple'_{\dataObject}(\attr_i) = \fctTypeCast{v_i}$, with $((\attr_1,v_1),\dots,(\attr_n,v_n)) \in X_\dataObject$, 
	$n = |\fctDom{\symAttrQueryMap}|$, and $\{\attr_1,\dots,\attr_n\} = \fctDom{\symAttrQueryMap}$.
	Considering $\symAttrQueryMap'$ is the restriction of $\symAttrQueryMap$ to $\symProjSet$,
	there exits a mapping tuple~$\mappingTuple''_{\dataObject} \in \sourceOp{\dataSource}{\queryExpr}{\symAttrQueryMap'}$,
	with Definition~\ref{def:source_operator}, such that $\fctDom{\mappingTuple''_\dataObject} = \symProjSet$, and 
	$\mappingTuple''_\dataObject(\attr) = \mappingTuple'_\dataObject(\attr)$ for all $\attr \in \symProjSet$. 
	Combining all the properties that we have shown, we can conclude that 
	$\fctDom{\mappingTuple} = \fctDom{\mappingTuple''_\dataObject} = \symProjSet$, and 
	$\mappingTuple(\attr) = \mappingTuple'_\dataObject(\attr) = \mappingTuple''_\dataObject(\attr)$ for all $\attr \in \symProjSet$
	and, thus, $\mappingTuple \in \sourceOp{\dataSource}{\queryExpr}{\symAttrQueryMap'}$; i.e, $\mappingTuple \in \symMappingInst'$.
	
	Lastly, we show that $\symMappingInst'  \subseteq \symMappingInst$ by considering an arbitrary mapping tuple 
    $\mappingTuple_\dataObject \in \symMappingInst'$ and prove that $\mappingTuple_\dataObject \in \symMappingInst$. 
	Given $\mappingTuple_\dataObject \in \symMappingInst'$ with Definition~\ref{def:source_operator}, and 
	$\fctDom{\symAttrQueryMap'} = \symProjSet$, it holds that 
	i)~$\dataObject \in \fctEval{\bigDataObject}{\queryExpr}$,
	ii)~$X_{\dataObject} = \bigtimes_{\attr \in \fctDom{\symAttrQueryMap'}}
		\big\{ (\attr, \var) \ \big| \
		\var \in \fctCbeval{\bigDataObject}{\dataObject}{\queryExpr'} \text{ with }  \queryExpr'\! = \symAttrQueryMap'(\attr) \big\}$, 
	iii)~$\fctDom{\mappingTuple_\dataObject} = \fctDom{\symAttrQueryMap'} = \symProjSet$, and
	iv)~$\mappingTuple_{\dataObject}(\attr_i) = \fctTypeCast{v_i}$, 
	with \\$((\attr_1,v_1),\dots,(\attr_n,v_n)) \in X_\dataObject$. 
	Now, provided with the conditions 
	$\fctDom{\symAttrQueryMap'} = \symProjSet$ and $\symProjSet \subseteq \fctDom{\symAttrQueryMap}$,
 	consider a mapping tuple $\mappingTuple'_\dataObject$
	for which it holds that 
	i)~$\fctDom{\mappingTuple'_\dataObject} = \fctDom{\symAttrQueryMap}$,
	ii)~$\mappingTuple'_\dataObject(\attr)$ is defined for all $\attr \in \fctDom{\symAttrQueryMap}$, and 
	iii)~$\mappingTuple_\dataObject'(\attr') = \mappingTuple_\dataObject(\attr')$ for all $\attr' \in \fctDom{\symProjSet}$.
	Thus, by Definition~\ref{def:source_operator}, $\mappingTuple'_\dataObject \in  \sourceOpDflt$. 	
	Next, let $\mappingTuple''$ be a mapping tuple that is the restriction of $\mappingTuple'_\dataObject$ to $\symProjSet$; 
	i.e, $\mappingTuple'' = \mappingTuple'_\dataObject[\symProjSet]$, which, by Definition~\ref{def:projection_operator}, 
	means that $\mappingTuple'' \in \fctProjectOpBig{\symProjSet}{ \sourceOpDflt }$.
	Combining everything together, we have that 
	i)~$\fctDom{\mappingTuple''} = \symProjSet = \fctDom{\mappingTuple_\dataObject}$, and 
	ii)~$\mappingTuple''(\attr) =\mappingTuple'_\dataObject(\attr) = \mappingTuple_\dataObject(\attr)$ for all $\attr \in \symProjSet$. 
	As a consequence, $\mappingTuple'' = \mappingTuple_\dataObject$ and, thus, 
	$\mappingTuple_\dataObject \in \fctProjectOpBig{\symProjSet}{\sourceOpDflt}$;i.e, $\mappingTuple_\dataObject \in \symMappingInst$.



	
	
    \qed
\end{proof}
