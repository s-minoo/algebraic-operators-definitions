\begin{algorithm}[t]
    \SetArgSty{textnormal}
    \DontPrintSemicolon
	\caption{Translates an RML mapping into our mapping algebra.}
	\label{alg:rml2algebra}
        \KwData{
            $\symRMLGraph$ - an RDF graph (assumed to contain a \emph{normalized} RML description) \\
            $\LTB$ - an injective function that maps all literals to blank nodes
        }
        \KwResult{ a mapping relation }

		$\omega_\mathsf{spog} \gets$ initially empty algebraic expression \\

		\ForAll{$\triplesMapIRI_\mathsf{tm}\! \in \symAllIRIs \cup \symAllBNodes$ for which there is a triple $(\triplesMapIRI_\mathsf{tm}, \ttlSmaller{rdf:type}, \ttlSmaller{rr:TriplesMap}) \in \symRMLGraph$}
        {

			$\omega \gets \sourceOp{\dataSource}{\queryExpr}{\symAttrQueryMap}$\!,
				%where
				with
			$(\dataSource, \queryExpr,\symAttrQueryMap) \!=\!$ \textsc{CreateSrcOpParam}($\triplesMapIRI_\mathsf{tm},\symRMLGraph$)
			\tcc*{Algorithm~\ref{alg:rml_source_op}} \\ 

			$u_\textsf{sm} \gets o$, where $o \in \symAllIRIs \cup \symAllBNodes$ such that $(\triplesMapIRI_\mathsf{tm},\ttlSmaller{rr:subjectMap},o) \in \symRMLGraph$ \\

			$\omega \gets \extOpAttr{\subjAttr\!}(\omega)$, where $\extExpr \!=\! \textsc{CreateExtExpr}(u_\textsf{sm}, \symRMLGraph, \LTB, \symAttrQueryMap)$
            \tcc*{Algorithm~\ref{alg:rml_termmap_extend_expr}} \\ 

			$u_\textsf{pm} \gets o$, where $o \in \symAllIRIs \cup \symAllBNodes$ such that $(\triplesMapIRI_\mathsf{tm},\ttlSmaller{rr:predicateMap},o) \in \symRMLGraph$ \\

			$\omega \gets \extOpAttr{\predAttr\!}(\omega)$, where $\extExpr \!=\! \textsc{CreateExtExpr}(u_\textsf{pm}, \symRMLGraph, \LTB, \symAttrQueryMap)$
            \tcc*{Algorithm~\ref{alg:rml_termmap_extend_expr}} \\

			$u_\mathsf{om} \gets o$, where $o \in \symAllIRIs \cup \symAllBNodes$ such that $(\triplesMapIRI_\mathsf{tm},\ttlSmaller{rr:objectMap},o) \in \symRMLGraph$ \\

			\If{there is a $u_\mathsf{ptm} \in \symAllIRIs \cup \symAllBNodes$ such that $(u_\mathsf{om},\ttlSmaller{rr:parentTriplesMap},u_\mathsf{ptm}) \in \symRMLGraph$}
            {

				$\symJoinAttrPairs \gets \emptyset$ \\
				\ForAll{$u_\mathsf{jc} \in \symAllIRIs \cup \symAllBNodes$ for which
					%there is a triple
				$(u_\mathsf{om},\ttlSmaller{rr:joinCondition},u_\mathsf{jc}) \in \symRMLGraph$}
                {

					$\attr \gets \symAttrQueryMap^{-1}(\lex)$, where $(u_\mathsf{jc},\ttlSmaller{rr:child},o) \in \symRMLGraph_\mathsf{tm}$ with $o = (\lex,\dt) \in \symAllLiterals$ \\

					$\attr'\! \gets \symAttrQueryMap^{-1}(\lex)$, where $(u_\mathsf{jc},\ttlSmaller{rr:parent},o) \in \symRMLGraph_\mathsf{tm}$ with $o = (\lex,\dt) \in \symAllLiterals$ \\

					$\symJoinAttrPairs \gets \symJoinAttrPairs \cup \{ (\attr, \attr') \}$ \\
                }

				$\omega \gets \fctEquiJoinOpBig{\omega}{ \sourceOp{\dataSource_\mathsf{ptm}}{\queryExpr_\mathsf{ptm}}{\symAttrQueryMap_\mathsf{ptm}} }$, where \par\hspace{15mm} $(\dataSource_\mathsf{ptm}, \queryExpr_\mathsf{ptm},\symAttrQueryMap_\mathsf{ptm}) =$ \textsc{CreateSrcOpParam}($u_\mathsf{ptm},\symRMLGraph$)
				\tcc*{Algorithm~\ref{alg:rml_source_op}} \\

				$u_\mathsf{sptm} \gets o$, where $o \in \symAllIRIs \cup \symAllBNodes$ such that $(u_\mathsf{ptm},\ttlSmaller{rr:subjectMap},o) \in \symRMLGraph$ \\

				$\omega \gets \extOpAttr{\objAttr\!}(\omega)$, where $\extExpr = \textsc{CreateExtExpr}(u_\mathsf{sptm}, \symRMLGraph, \LTB, \symAttrQueryMap_\mathsf{ptm})$
            }
            \Else{
				$\omega \gets \extOpAttr{\objAttr\!}(\omega)$, where $\extExpr = \textsc{CreateExtExpr}(u_\textsf{om}, \symRMLGraph, \LTB, \symAttrQueryMap)$
            }

			$u_\textsf{pom} \gets o$, where $o \in \symAllIRIs \cup \symAllBNodes$ such that $(\triplesMapIRI_\mathsf{tm},\ttlSmaller{rr:predicateObjectMap},o) \in \symRMLGraph$ \\ 

			\If{there is a $u_\mathsf{gm} \in \symAllIRIs \cup \symAllBNodes$ such that
            $(u_\textsf{pom},\ttlSmaller{rr:graphMap},u_\mathsf{gm}) \in \symRMLGraph$  or $(u_\textsf{sm},\ttlSmaller{rr:graphMap},u_\mathsf{gm}) \in \symRMLGraph$}
            {

				$\omega \gets \extOpAttr{\graphAttr\!}(\omega)$, where $\extExpr = \textsc{CreateExtExpr}(u_\mathsf{gm}, \symRMLGraph, \LTB, \symAttrQueryMap)$
            }
            \Else{
				$\omega \gets \extOpAttr{\graphAttr\!}(\omega)$, where $\extExpr$ is the IRI \ttlSmaller{rr:defaultGraph}
            }

			$\omega_\mathsf{spog} \gets \fctUnionBig{\omega_\mathsf{spog}}{ \fctProjectOp{ \{\subjAttr,\predAttr,\objAttr,\graphAttr\} }{\omega} }$
        }
        \Return{$\omega_\mathsf{spog}$}
\end{algorithm}
