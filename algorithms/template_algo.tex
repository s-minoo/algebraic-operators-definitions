	\caption{\textsc{Template} -
		%creates an extend expression for resolving a string template.}
		%creates an extend expression to resolve a string template.}
		%creates extend expressions for resolving string templates.}
		%creates extend expressions to resolve string templates.}
		creates extend expressions to instantiate string templates.}
	\label{alg:template_function}
        \KwData{%
            \textit{tmpl} - a string (assumed to be an RML string template) \;
            \hspace{10mm}
            $\symAttrQueryMap$ - an injective partial function that maps attributes to query expressions
        }
        \KwResult{an extend expression (as per Definition~\ref{def:ExtendExpressions:Syntax})}
        $\substrSeq \gets \fctTempSubStrs{\textit{tmpl}}$ \tcp*[r]{Definition~\ref{def:tempSubStrs}}
        $\extExpr \gets (s_1, \ttlSmaller{xsd:string})$, where $s_1$ is the first element in $\substrSeq$ \;
        \ForEach(\tcp*[f]{Iterate
        	%based on the order
        	using order
        in~$\substrSeq'$}) {$s_i$ in $\substrSeq'\!$, where $\substrSeq'$ is $\substrSeq$ without the first element}
        {
            \If{$s_i$ is a string that starts with "\{" and ends with "\}"}
            {
                $\extExpr \gets \fctConcatBig{\extExpr}{\symAttrQueryMap^{-1}(\queryExpr)}$, where $\queryExpr$ is
	                  %the substring of
                $s_i$ without
                    	the
                enclosing "\{" and "\}" \;
            }
            \Else{
                $\extExpr \gets \fctConcatBig{\extExpr}{(s_i, \ttlSmaller{xsd:string})}$ \;
            }
        }
        \Return{$\extExpr$}
