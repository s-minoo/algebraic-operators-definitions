\subsection{Projection Operator}%
\label{sub:projection_operator}

\begin{definition}
	Given a mapping relation $\mappingRel$, with $\schema = \schemaTuple$, and
	a \emph{non-empty} set of variables $\symProjSet \subset \symAttrSubSet$.
	The \textbf{projection} operator changes the schema of the mapping
	relation, $\schemaTuple$,  by restricting  attributes in $\symAttrSubSet$ to those in
	$\symProjSet \cap \symAttrSubSet$. In other words, it creates a new mapping relation
	with the projected schema, $\schema_{projected}$ that is represented by the
	tuple $(\symProjSet \cap \symAttrSubSet, \fragAttr)$
	\label{def:projection}
\end{definition}

\begin{equation}
	\begin{split}
		\text{Project}(\schema, \symProjSet)     & = ( \symProjSet \cap \symAttrSubSet , \fragAttr)        \\
		\text{Project}(\mappingRel, \symProjSet) & =  (\text{Project}(\schema, \symProjSet), \mappingInst)
	\end{split}
	\label{eq:projection}
\end{equation}



\subsection{Rename Operator}%
\label{sub:rename_operator}

\begin{definition}
	\label{def:rename}
	Given a set of pairs of variables $\{(a_{a1}, a_{b1})..(a_{an},a_{bn})\}
		\in \symRenameSet$ where $a_n \in \symAttrUniverse$.
	The \textbf{rename} operator, applied on a mapping relation $\mappingRel$,
	renames the attributes of the schema of the mapping relation, $\schema =
		\schemaTuple$ using the attribute pairs in $\symRenameSet$.
	It creates a mapping relation, with a new schema containing a new non-empty
	attribute set $\symAttrSubSet_{renamed}$ as follows:
	$\symAttrSubSet_{renamed} = \{ a_{new} \ | \  \forall a \in \symAttrSubSet.
		\forall (a_{old}, a_{new}) \in \symRenameSet.\ a = a_{old} \} $
	If the rename operator is configured with an alias string $s_{alias}$
	instead of $\symRenameSet$, the rename operator will concatenate
	$s_{alias}$ as the prefix $\forall a \in \symAttrSubSet$ where $s_{alias} \
		\Vert \ a \notin \symAttrSubSet$ as $s_{alias}\ \Vert \ a$.

\end{definition}

\begin{equation}
	\begin{split}
		\text{Rename}(\symAttrSubSet, \symRenameSet) & =
		\{ a_{new} \ | \  \forall a \in \symAttrSubSet.
		\forall (a_{old}, a_{new}) \in \symRenameSet.\ a = a_{old} \}                                              \\
		\text{Rename}(\schema, \symRenameSet)        & = (\text{Rename}(\symAttrSubSet, \symRenameSet), \fragAttr) \\
		\text{Rename}(\mappingRel, \symRenameSet)    & = (\text{Rename}(\schema, \symRenameSet), I)
	\end{split}
	\label{eq:rename}
\end{equation}

\begin{equation}
	\begin{split}
		\text{Rename}(\symAttrSubSet, s_{aliased}) & = \{ s_{aliased} \ \Vert \ a \ |\ \forall a \in \symAttrSubSet.\ s_{aliased} \ \Vert \ a \notin \symAttrSubSet \} \\
		\text{Rename}(\schema, s_{aliased})        & = (\text{Rename}(\symAttrSubSet, s_{alaised}), \fragAttr)                                                        \\
		\text{Rename}(\mappingRel, s_{aliased})    & =  (\text{Rename}(\schema, s_{aliased}), \mappingInst)
	\end{split}
	\label{eq:rename_alias}
\end{equation}
