\section{Algebraic Mapping Operators}%
\label{sec:algebraic_mapping_operators}

\subsection{Source Operator}%
\label{sub:source_operator}

\subsubsection{Notations}%
\label{ssub:notations}

We assume the following pairwise disjoint countably infinite sets:
$\symDatasetUni$ (datasets), $\symDataObjUni$ (data objects) and
$\symDataAccUni$ (data access languages).
Each dataset in $\symDatasetUni$ is a collection of data objects.
To denote that a data object~$\dataObject \in \symDataObjUni$ is contained in
such a dataset~$\symDataset \in \symDatasetUni$, we write $\dataObject \in
	\symDataset$.
As for the languages in $\symDataAccUni$, we consider each such
language~$\symDataAcc \in \symDataAccUni$ as a set where every
element~$\queryExpr \in \symDataAcc$ is one of the expressions of the language.

Informally, a data object~$\dataObject \in \symDataObjUni$ is an abstract
entity that could be queried using a query expression to extract
one or more data objects \textbf{which can, in turn, be accessible by different data access languages}.
For example, a JSON object is considered a data object with JSONPath as the data
access language, and it can be further queried using JSONPath expressions to
extract JSON arrays (data objects) or a JSON field containing a string
representation of CSV rows (data objects), which can be further queried using a CSV specific
data access language.





\subsubsection{Source Type}%
\label{ssub:source_type}

\begin{definition}
	\label{def:source_type}
	A source type is a tuple $\sourceType = \sourceTypeTuple$ such that
	$\symSetDataset \subseteq \symDatasetUni$, $\symDataAcc \in \symDataAccUni$,
	$\eval$ is a partial function $\eval: \symSetDataset \times \symDataAcc \rightarrow \symDataSeq$,
	$\cbeval$ is a partial function $\cbeval: \symSetDataset \times \symDataObjUni \times \symDataAcc \rightarrow \symDataSeq $
	where $\symDataSeq$ is the set of all possible sequences of data objects in $\symDatasetUni$.
\end{definition}


\begin{definition}
	\label{def:data_source}
	A data source is a tuple~$\dataSource = \dataSourceTuple$ such that $\symDataset \in \symSetDataset$
	and \break $\sourceType = \sourceTypeTuple$.
\end{definition}


\begin{definition}
	\label{def:cast_function}
	A \textbf{type-cast} function, $\typeCast$ is a partial function
	$\typeCast: \symDataObjUni \rightarrow \symAllLiterals$.
    For every sequence $\symDataSeq$ of data objects, we write
    $\typeCast[\symDataSeq]$ to denote the sequence of literals and error
    values~$\error$ such that the $i$-th element in $\typeCast[\symDataSeq]$
    is $\fctTypeCast{\dataObject_i}$, if $\dataObject_i \in dom(\typeCast)$, 
    and $\error$ otherwise, where $\dataObject_i$ is the $i$-th element in 
    $\symDataSeq$, for all $i \in  \{1,\dots,|\symDataSeq|\}$.
    %we denote
	%$\typeCast[\symDataSeqCard{m}] = \{ \gamma_i \ | \ \dataObject_i \in \symDataSeqCard{m}.\ \gamma_i = \fctTypeCast{\dataObject_i} \}_{i=1}^{m}$
	%provided that $\symDataSeqCard{m}$ is a sequence, with
	%$|\symDataSeqCard{m}| = m$, of data objects.
	The order of the data objects is preserved.
\end{definition}

\begin{definition}
	\label{def:source_operator}
    Let $\dataSource = \dataSourceTuple$ be a data source with
    $\sourceType=\sourceTypeTuple$,
    $\symAttrQueryPairs \subseteq \symAttrSubSet \times \symDataAcc$ be a set
    of pairs $\attrQueryPair \in \symAttrQueryPairs$,
    $\queryExpr_{root} \in \symDataAcc$,
	and the source configurations be a tuple~$\iterConfig = \iterConfigTuple$.
	The \emph{generation of mapping relations from the source~$\dataSource$ configured
		with $\iterConfig$}, denoted by
	$\fctSourceOp{\dataSource}{\iterConfig}$, is the mapping relation
	$\mappingRel$ derived as follows:
\end{definition}



\begin{equation}
	\begin{split}
		\symAttrSubSet           & = \{ \attr \ | \ \forall \attrQueryPair \in \symAttrQueryPairs\} \cup \{\fragAttr\}                                                                                                                                    \\
		\symDataSeq_{root}       & = (\dataObject_i)_{i=1}^{n} = \fctEval{\symDataset}{\queryExpr_{root}}                                                                                                                                                 \\
		\mappingInst^{map}       & = \{(a, \Gamma, i) \ | \ \dataObject_i \in \symDataSeq_{root}.\  \forall \attrQueryPair \in \symAttrQueryPairs.\ \symDataSeq = \fctCbeval{\symDataset}{\dataObject_i}{\queryExpr}.\ \Gamma =  \typeCast[\symDataSeq]\} \\
		\mappingInst^{map}[a, i] & = \{\Gamma \ | \ (a, \Gamma, i) \in \mappingInst^{map}\}                                                                                                                                                               \\
		\mappingInst^{flattened} & =  \{ (a, \times \I) \text{???} \}
	\end{split}
	\label{eq:source_operator}
\end{equation}





\subsection{Projection Operator}%
\label{sub:projection_operator}

The \textbf{projection} operator restricts the mapping tuples in a mapping
relation, $\mappingRel$, to a set of attributes $\symProjSet \subseteq
	\symAttrSubSet$ as defined in Definition~\ref{def:tuple_restrict}.

\begin{definition}
	\label{def:projection}
	Let $r = \mappingRel$ be a mapping relation and $\symProjSet \subseteq
		\symAttrSubSet \setminus \{\fragAttr\}$ be a non-empty subset of the
	attributes of $r$.
	The \emph{$\symProjSet$-specific projection of $r$}, denoted by
	$\text{Project}^\symProjSet(r)$, is the mapping relation $(\symAttrSubSet',
		\mappingInst')$ such that $\symAttrSubSet' = \symProjSet \cup
		\{\fragAttr\}$ and $\mappingInst' = \{ \mappingTuple[\symAttrSubSet'] \ | \
		\mappingTuple \in \mappingInst \}$.
\end{definition}

\begin{equation}
	\begin{split}
		\mappingInst'                             & =
		\{\mappingTuple' \ | \ \forall \mappingTuple \in \mappingInst.\ \mappingTuple[\symProjSet \cup\{\fragAttr\}] = \mappingTuple' \} \\
		\symAttrSubSet'                           & =  \symProjSet \cup \{\fragAttr\}                                                    \\
		\text{Project}^{\symProjSet}(\mappingRel) & =  ( \symAttrSubSet', \mappingInst' )
	\end{split}
	\label{eq:projection}
\end{equation}



\subsection{Rename Operator}%
\label{sub:rename_operator}
Given a mapping relation $\mappingRel$, and set of pairs of attributes
$\symRenameSet$ with $\{(a_{1}, b_{1}),..(a_{n}, b_{n})\}$ where all $a$'s and
$b$'s are unique.
A \textbf{rename operator} renames the attributes in the mapping relations,
$\mappingRel$,  according to set of the pairs of attributes, $\symRenameSet$.
It creates a new mapping relation $(\symAttrSubSet', \mappingInst')$
such that $\symAttrSubSet'$ is a set of attributes where,
given a pair of attributes $(a_{old},b) \in \symRenameSet$, all
attributes $a \in \symAttrSubSet$ is replaced with $b$ if $a = a_{old}$ otherwise
$a$ is kept in $\symAttrSubSet'$.
For the \emph{renamed mapping tuples}~$\mappingTuple'$, it is the result of
restricting the mapping tuple~$\mappingTuple$ to $\symAttrSubSet'$ as
$\mappingTuple[\symAttrSubSet']$ with the condition that for each
pair $(a,b) \in \symRenameSet$, $\mappingTuple(b) = \mappingTuple(a)$, thus,
$\mappingTuple'$ is defined for all attributes in $\symAttrSubSet'$.

\begin{definition}
	Let $r = \mappingRel$ be a mapping relation, $\symRenameSet = \{(a,b) \in \symAttrSubSet \times \symAttrUniverse : a \neq  b \neq \fragAttr\}$
	a non-empty countably finite set of pairs of attributes,
	$\symRenameSet_{first} = \{ a \ | \ \forall (a,b) \in \symRenameSet \}$.
	The \emph{$\symRenameSet$-specific rename of $r$}, denoted by $\text{Rename}^R(r)$,
	is the mapping relation $(\symAttrSubSet', \mappingInst')$ such that:
	$\symAttrSubSet'=  \{ a \ | \  a \in \symAttrSubSet.\ a \notin
		\symRenameSet_{first} \}  \cup \{ b \ | \ \forall a  \in \symAttrSubSet.\
		\exists (a_{old}, b) \in \symRenameSet.\ a = a_{old}\}$.
	$\mappingTuple[\symAttrSubSet']$ is such that for~all~$a_{new} \in
		\symAttrSubSet'$, if there exists $(a_{old},b) \in \symRenameSet$ and $a_{new} =
		b$, then $\mappingTuple(b) = \mappingTuple(a_{old})$.
	$\mappingInst' = \{ \mappingTuple' \ | \ \mappingTuple \in \mappingInst.\
		\mappingTuple' = \mappingTuple[\symAttrSubSet'] \}$.
\end{definition}



\begin{equation}
	\begin{split}
		\symAttrSubSet'                & = \{ a \ | \  a \in \symAttrSubSet \setminus \symRenameSet_{first} \}  \cup \{ b \ | \ \forall a  \in \symAttrSubSet.\ \exists (a_{old}, b) \in \symRenameSet.\ a = a_{old}\} \\
		\mappingTuple[\symAttrSubSet'] & = \left\{
		\begin{array}{rcl}
			\mappingTuple(a)       & \mbox{for} & \forall a \in \symAttrSubSet \setminus \symRenameSet_{first}                                \\
			\mappingTuple(a_{old}) & \mbox{for} & \forall a_{new} \in \symAttrSubSet'.\ \exists (a_{old}, b) \in \symRenameSet.\  a_{new} = b
		\end{array}
		\right.                                                                                                                                                                                                        \\
		\mappingInst'                  & = \{\mappingTuple[\symAttrSubSet'] \ | \ \mappingTuple \in \mappingInst\}                                                                                                     \\
		\text{Rename}_{R}(\mappingRel) & = (\symAttrSubSet', \mappingInst')
	\end{split}
	\label{eq:rename}
\end{equation}
