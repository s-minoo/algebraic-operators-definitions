\subsection{Source Operator}%

\subsubsection{Notations}%
\label{ssub:source_notations}

We assume the following pairwise disjoint countably infinite sets:
$\symDatasetUni$ (datasets), $\symDataObjUni$ (data objects) and
$\symDataAccUni$ (data access languages).

Each dataset in $\symDatasetUni$ is a collection of data objects.
To denote that a data object~$\dataObject \in \symDataObjUni$ is contained in
such a dataset~$\symDataset \in \symDatasetUni$, we write $\dataObject \in
	\symDataset$.
As for the languages in $\symDataAccUni$, we consider each such
language~$\symDataAcc \in \symDataAccUni$ as a set where every
element~$\queryExpr \in \symDataAcc$ is one of the expressions of the language.

Informally, a data object~$\dataObject \in \symDataObjUni$ is an abstract
entity that could be queried using a query expression to extract
one or more data objects \textbf{which can, in turn, be accessible by different data access languages}.
For example, a JSON object is considered a data object with JSONPath as the data
access language, and it can be further queried using JSONPath expressions to
extract JSON arrays (data objects) or a JSON field containing a string
representation of CSV rows (data objects), which can be further queried using a CSV specific
data access language.

\subsection{Projection Operator}%
\label{sub:projection_operator}

The \textbf{projection} operator restricts the mapping tuples in a mapping
relation, $\mappingRel$, to a set of attributes $\symProjSet \subseteq
	\symAttrSubSet$ as defined in Definition~\ref{def:tuple_restrict}.

\begin{definition}
	\label{def:projection_operator}
	Let $r = \mappingRel$ be a mapping relation and $\symProjSet \subseteq
		\symAttrSubSet \setminus \{\fragAttr\}$ be a non-empty subset of the
	attributes of $r$.
	The \emph{$\symProjSet$-specific projection of $r$}, denoted by
	$\text{Project}^\symProjSet(r)$, is the mapping relation $(\symAttrSubSet',
		\symMappingInst')$ such that $\symAttrSubSet' = \symProjSet \cup
		\{\fragAttr\}$ and $\symMappingInst' = \{ \mappingTuple[\symAttrSubSet'] \ | \
		\mappingTuple \in \symMappingInst \}$.
\end{definition}




\subsection{Rename Operator}%
\label{sub:rename_operator}
Given a mapping relation $\mappingRel$, and set of pairs of attributes
$\symRenameSet$ with $\{(a_{1}, b_{1}),..(a_{n}, b_{n})\}$ where all $a$'s and
$b$'s are unique.
A \textbf{rename operator} renames the attributes in the mapping relations,
$\mappingRel$,  according to set of the pairs of attributes~$\symRenameSet$.
It creates a new mapping relation $(\symAttrSubSet', \symMappingInst')$
such that $\symAttrSubSet'$ is a set of attributes where,
given a pair of attributes $(a_{old},b) \in \symRenameSet$, for all
attributes $a \in \symAttrSubSet$ is replaced with $b$ if $a = a_{old}$ otherwise
$a$ from is kept in $\symAttrSubSet'$.
For the \emph{renamed mapping tuples}~$\mappingTuple'$, it is the result of
restricting the mapping tuple~$\mappingTuple$ to $\symAttrSubSet'$ as
$\mappingTuple[\symAttrSubSet']$ with the condition such that for each
pair $(a_{old},b) \in \symRenameSet$, we have $\mappingTuple'(b) = \mappingTuple(a_{old})$,
and for every $a$ in $A$ that did not get renamed, we have $\mappingTuple'(a) =
	\mappingTuple(a)$, thus, $dom(\mappingTuple') = \symAttrSubSet'$.

\begin{definition}
	\label{def:rename_operator}
	Let $r = \mappingRel$ be a mapping relation, $\symRenameSet = \{(a,b) \in \symAttrSubSet \times \symAttrUniverse : a \neq  b \neq \fragAttr\}$
	a non-empty countably finite set of pairs of attributes, and
	$\symRenameSet_{first} = \{ a \ | \ (a,b) \in \symRenameSet\}$.
	The \emph{$\symRenameSet$-specific rename of $r$}, denoted by $\text{Rename}^R(r)$,
	is the mapping relation $(\symAttrSubSet', \symMappingInst')$ such that
	\begin{equation*}
		\symAttrSubSet'  =  (\symAttrSubSet \setminus \symRenameSet_{first}) \cup \{ c \ | \ a \in \symAttrSubSet \text{ and } (b,c) \in \symRenameSet \text{ such that } a = b\}
	\end{equation*}
	and
	\begin{equation*}
		\symMappingInst' = \{ \mappingTuple' \ | \
		\mappingTuple \in \symMappingInst \text{ and }
		a \in \symAttrSubSet' \text{ and }
		(b, c) \in \symRenameSet \text{ such that }
		\text{if } a = c,\ \mappingTuple'(a) = \mappingTuple(b)
		\text{ else } \mappingTuple'(a) = \mappingTuple(a)
		\}
	\end{equation*}
\end{definition}

\subsection{Extend Operator}%
\label{sub:extend_operator}

\subsubsection{Extend expressions}%
\label{ssub:extend_expressions}
We assume a countably infinite set $\symExtExprUni$ (extend expressions) which is
pairwise disjoint with the countably infinite sets defined in
Section~\ref{ssub:source_notations} and Section~\ref{sec:preliminaries}.
We also denote $\symExtExprSeqUni$ as a countably infinite set of all possible
sequences of \emph{extend expressions} in $\symExtExprUni$.

\begin{definition}
	\label{def:extend_expression}
	Given a mapping instance $\symMappingInst$.
	An element $\extExpr \in \symExtExprUni$ is an expression which can be
	evaluated by the partial functions
	\begin{equation*}
		\extExprTupleEval : \symExtExprUni \times \symMappingInst \rightarrow \symTermSeqUniverse \cup \{ \error \}
	\end{equation*}
	and
	\begin{equation*}
		\extExprInEval: \symExtExprUni \times (\symTermSeqUniverse \cup \{\error\}) \rightarrow \symTermSeqUniverse \cup \{\error\}
	\end{equation*}
	where $\symTermSeqUniverse$ is a set of all possible sequences of terms in $\symTermUniverse$.
	We also denote an evaluation of expression sequence~$\symExtExprSeq$ on
	a mapping tuple~$\mappingTuple$ as a partial function
	$\extEvalSeq: \symExtExprSeqUni \times \symMappingInst \rightarrow \symTermSeqUniverse \cup \{\error\}$
	such that
	\begin{equation*}
		\fctExtEvalSeq{\symExtExprSeq}{\mappingTuple} =
		\fctExtExprInEval{\extExpr_n}{
			\fctExtExprInEval{\dots}{
				\fctExtExprInEval{\extExpr_2}{
					\fctExtExprTupleEval{\extExpr_1}{\symMappingInst}
				}
			}
		}
	\end{equation*}
	where $\extExpr_i$ is the $i$-th expression element in $\symExtExprSeq$
	for every $i \in \{1,\dots,|\symExtExprSeq|\}$.
\end{definition}


\begin{definition}
	\label{def:extend_operator}
	Let $r = \mappingRel$ be a mapping relation, a set of pairs $\symExtPairs: \extPairsType$.
	The \emph{$\symExtPairs$-specific extension of $r$} is a
	mapping relation $(\symAttrSubSet', \symMappingInst')$ such that
	$\symAttrSubSet' = \symAttrSubSet \cup \{ a \ | \ (a, \symExtExprSeq) \in \symExtPairs\}$
	and
	\begin{equation*}
		\symMappingInst' =  \left\{ \{a_1 \rightarrow v_1, \dots,a_n \rightarrow v_n \} \ |\
		\mappingTuple \in \symMappingInst \text{ and }
		((a_1 \rightarrow v_1), \dots,(a_n \rightarrow v_n) ) \in X_{\mappingTuple}
		\right\}
	\end{equation*}
	where
	\begin{equation*}
		X_{\mappingTuple}=  \bigtimes_{\attr \in \symAttrSubSet'} \symMappingInst_{\mappingTuple}^{\attr}
	\end{equation*}
	such that
	\begin{equation*}
		\symMappingInst_{\mappingTuple}^{\attr} = \{ (a \rightarrow v) \ | \
		\text{ if } \attr \in dom(\mappingTuple) \text{ then } v = \mappingTuple(a)
		\text{ else } (\attr, \symExtExprSeq) \in \symExtPairs \text{ and } v \in \fctExtEvalSeq{\symExtExprSeq}{\mappingTuple}
		\}
	\end{equation*}


\end{definition}




\subsection{Fragmenter Operator}%
\label{sub:fragmenter_operator}


\begin{definition}
	\label{def:fragmenter_operator}
	Let $r=\mappingRel$ be a mapping relation, and a set of pairs
	$\symFragmentPairs: \symFragmentUniverse \times \symFragmentUniverse$.
	The \emph{$\symFragmentPairs$-specific fragmentation of $r$} is
	a mapping relation $(\symAttrSubSet', \symMappingInst')$ such that
	$\symAttrSubSet' =  \symAttrSubSet$ and
	\begin{equation*}
		\symMappingInst'= \symMappingInst \cup \{ \mappingTuple' \ | \ \mappingTuple \in \symMappingInst \text{ and }
		(\mappingTuple(\fragAttr), f_{new}) \in \symFragmentPairs \text{ and }
		\attr \in \symAttrSubSet \setminus \{\fragAttr\} \text{ such that }
		\mappingTuple'(\attr) = \mappingTuple(\attr) \text{ and }
		\mappingTuple'(\fragAttr) = f_{new} \}
	\end{equation*}
\end{definition}


\subsection{$\theta$-join Operator}%
\label{sub:theta_join_operator}

\begin{definition}
	\label{def:theta_join_operator}
	Given two mapping relations $r_{left} = \mappingRelSpec{left}$ and
	$r_{right} = \mappingRelSpec{right}$, and a set of pairs of attributes
	$\symJoinAttrPairs: \symAttrSubSet_{left} \times \symAttrSubSet_{right}$.
	We assume that $\symAttrSubSet_{left} \cap \symAttrSubSet_{right} = \{\fragAttr\}$.
	The \emph{$\symJoinAttrPairs$-specific join of mapping relations $r_{left}$ and
		$r_{right}$} is the mapping relation $(\symAttrSubSet', \symMappingInst')$
	such that $\symAttrSubSet' = \symAttrSubSet_{left} \cup \symAttrSubSet_{right}$
	and
	\begin{equation*}
		\symMappingInst' =  \{ \mappingTuple' \cup \mappingTuple'' \ | \
		\mappingTuple' \in \symMappingInst_{left} \text { and }
		\mappingTuple'' \in \symMappingInst_{right} \text{ such that }
        \mappingTuple'(\fragAttr) = \mappingTuple''(\fragAttr) \text { and }
        \mappingTuple'(\attr') = \mappingTuple''(\attr'') \text{ for all }
        (\attr',\attr'') \in \symJoinAttrPairs
		\}
	\end{equation*}
\end{definition}
