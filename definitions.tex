\section{Algebraic Mapping Operators}%
\label{sec:algebraic_mapping_operators}

\subsection{Source Operator}%
\label{sub:source_operator}
A mapping process maps heterogeneous datasets to RDF graphs.
Thus, an operator needs to be defined for the generation of mapping
relations~(Definition~\ref{def:mapping_relation}) from heterogeneous datasets
containing heterogeneous \emph{data objects}.
Data objects can be a row of CSV record or a relational table from a relational
database.

The first step in the source operator is the generation of data objects
from the input dataset with \emph{iterators}.
The input dataset is first queried using the relevant \emph{data access
	language}, with the \emph{root iterator}, to extract a set of data objects
from the input dataset.
For example in Table~\ref{tab:nested_data_objects}, a CSV root iterator
generates rows of CSV records with the columns \emph{name}, and \emph{age}.

The root iterator provides the context upon which, the subsequent \emph{context-based
	object} iterators could be executed for further querying and extraction of data
objects, using a data access language.
For example, given the input data objects in
Table~\ref{tab:nested_data_objects}, where two CSV data objects  are generated
by a CSV root iterator.
Notice that the CSV data objects contain JSON data objects in the column
\emph{contact-info}.
Applying a context-based object iterator with the JSONPath
\emph{\$.contact-info.lastName}, results in the generation of the data objects
shown in Table~\ref{tab:json_extracted_data_object}.

After extracting the data objects, the extracted data objects are each paired
with an attribute $a$ in $\symAttrUniverse$, and then \emph{type-casted} using
a \emph{casting function} to RDF \emph{literals}~$l$ in $\symAllLiterals$.
The set containing the pairs~$(a,l)$ and a default \emph{fragment pair}~$(\fragAttr, f_0)$
results in the mapping tuples~$\mappingTuple$ such that the generated mapping
relation $\mappingRel$ with the mapping tuples~$\mappingTuple$ in $\mappingInst$ and
all the $a$ in $dom(\mappingTuple)$ as the elements of the set $\symAttrSubSet$.

In order to make the definition of the source operator more accessible for the
readers, we first define the smallest and the most important component of the
source operator: \textbf{iterators}.


\subsubsection{Notations}%
\label{ssub:notations}

We assume the following two pairwise disjoint countably infinite sets:
$\symDatasetUni$ (datasets), and  $\symDataAccUni$ (data access languages).
A set of data objects~$\symDataset$ is subset of $\symDatasetUni$, with
$\dataObject$ a data object in $\symDataset$, and a query
expression~$\queryExpr$ is an element of $\symDataAccUni$.
We also define an evaluation function $\eval(\dataObject, \queryExpr)$ to
denote the evaluation of the query expression $\queryExpr$ on the given data
object $\dataObject$ for which $\queryExpr$ is a valid expression for querying 
$\dataObject$.
The result of such an evaluation is a finite sequence of data objects denoted
as $\symDataSeq$ which can be empty.


\subsubsection{Iterators}%
\label{ssub:iterators}
Given a finite set of data objects~$\symDataset$ (a subset of $\symDatasetUni$)
and a finite non-empty set of data access language $\symDataAcc$ (a subset of
$\symDataAccUni$).
An \textbf{iterator} queries a set of data objects, $\symDataset$, using a
query expression~$\queryExpr$ in $\symDataAcc$ to extract a \emph{sequence} of
data objects, with the requirement that $\queryExpr$ is defined for querying
data objects in $\symDataset$.
There are two types of iterators defined in this work: i) \emph{root iterator},
and ii) \emph{context-based object iterator}.
We define them respectively as follows: 

\begin{definition}
	\label{def:root_iterator}
	Given a set of data objects~$\symDataset$, and a query expression $\queryExpr$
	defined for querying $\symDataset$.
	We write $\fctRootIt{\symDataset}{\queryExpr}$ to denote the
	evaluation of $\eval(\symDataset, \queryExpr)$ such that
    a finite set of data objects~$\symDataset^{context}$ is generated, and  $\symDataset^{context} =
        \{\dataObject_1, ... , \dataObject_n\}$, where $\dataObject_n$ is the
    $n^{th}$-iteration of evaluating $\eval(\symDataset, \queryExpr)$.
\end{definition}


\begin{definition}
	\label{def:cob_iterator}
	Given a set of data objects~$\symDataset$, a context data
	object~$\dataObject \in \symDataset$, and a query expression $\queryExpr$
	defined for accessing the data objects in~$\symDataset$.
	We write $\fctCobit{\symDataset}{\dataObject}{\queryExpr}$ to denote
    the evaluation of 
    $\eval(\dataObject, \queryExpr)\ ||\ \forall i \in \symDataset \setminus \{ \dataObject \} \ \eval(i, \queryExpr)$
    such that a sequence of data objects $\symDataSeq$ is generated, where 
    $\symDataSeq =  \symDataSeq_{context} \ || \ \symDataSeq_{relative}$
    provided that $\symDataSeq_{context} = \eval(\dataObject, \queryExpr)$, and 
    $\symDataSeq_{relative} = \forall i \in \symDataset \setminus \{ \dataObject \} \ \eval(i, \queryExpr)$.
    We denote $\symDataSeq_{context} \ || \ \symDataSeq_{relative}$ as appending the 
    two sequences $\symDataSeq_{context}$ and $\symDataSeq_{relative}$ in order.

\end{definition}

\subsection{Projection Operator}%
\label{sub:projection_operator}

The \textbf{projection} operator restricts the mapping tuples in a mapping
relation, $\mappingRel$, to a set of attributes $\symProjSet \subseteq
	\symAttrSubSet$ as defined in Definition~\ref{def:tuple_restrict}.

\begin{definition}
	\label{def:projection}
	Let $r = \mappingRel$ be a mapping relation and $\symProjSet \subseteq
		\symAttrSubSet \setminus \{\fragAttr\}$ be a non-empty subset of the
	attributes of $r$.
	The \emph{$\symProjSet$-specific projection of $r$}, denoted by
	$\text{Project}^\symProjSet(r)$, is the mapping relation $(\symAttrSubSet',
		\mappingInst')$ such that $\symAttrSubSet' = \symProjSet \cup
		\{\fragAttr\}$ and $\mappingInst' = \{ \mappingTuple[\symAttrSubSet'] \ | \
		\mappingTuple \in \mappingInst \}$.
\end{definition}

\begin{equation}
	\begin{split}
		\mappingInst'                             & =
		\{\mappingTuple' \ | \ \forall \mappingTuple \in \mappingInst.\ \mappingTuple[\symProjSet \cup\{\fragAttr\}] = \mappingTuple' \} \\
		\symAttrSubSet'                           & =  \symProjSet \cup \{\fragAttr\}                                                    \\
		\text{Project}^{\symProjSet}(\mappingRel) & =  ( \symAttrSubSet', \mappingInst' )
	\end{split}
	\label{eq:projection}
\end{equation}



\subsection{Rename Operator}%
\label{sub:rename_operator}
Given a mapping relation $\mappingRel$, and set of pairs of attributes
$\symRenameSet$ with $\{(a_{1}, b_{1}),..(a_{n}, b_{n})\}$ where all $a$'s and
$b$'s are unique.
A \textbf{rename operator} renames the attributes in the mapping relations,
$\mappingRel$,  according to set of the pairs of attributes, $\symRenameSet$.
It creates a new mapping relation $(\symAttrSubSet', \mappingInst')$
such that $\symAttrSubSet'$ is a set of attributes where,
given a pair of attributes $(a_{old},b) \in \symRenameSet$, all
attributes $a \in \symAttrSubSet$ is replaced with $b$ if $a = a_{old}$ otherwise
$a$ is kept in $\symAttrSubSet'$.
For the \emph{renamed mapping tuples}~$\mappingTuple'$, it is the result of
restricting the mapping tuple~$\mappingTuple$ to $\symAttrSubSet'$ as
$\mappingTuple[\symAttrSubSet']$ with the condition that for each
pair $(a,b) \in \symRenameSet$, $\mappingTuple(b) = \mappingTuple(a)$, thus,
$\mappingTuple'$ is defined for all attributes in $\symAttrSubSet'$.

\begin{definition}
	Let $r = \mappingRel$ be a mapping relation, $\symRenameSet = \{(a,b) \in \symAttrSubSet \times \symAttrUniverse : a \neq  b \neq \fragAttr\}$
	a non-empty countably finite set of pairs of attributes,
	$\symRenameSet_{first} = \{ a \ | \ \forall (a,b) \in \symRenameSet \}$.
	The \emph{$\symRenameSet$-specific rename of $r$}, denoted by $\text{Rename}^R(r)$,
	is the mapping relation $(\symAttrSubSet', \mappingInst')$ such that:
	$\symAttrSubSet'=  \{ a \ | \  a \in \symAttrSubSet.\ a \notin
		\symRenameSet_{first} \}  \cup \{ b \ | \ \forall a  \in \symAttrSubSet.\
		\exists (a_{old}, b) \in \symRenameSet.\ a = a_{old}\}$.
	$\mappingTuple[\symAttrSubSet']$ is such that for~all~$a_{new} \in
		\symAttrSubSet'$, if there exists $(a_{old},b) \in \symRenameSet$ and $a_{new} =
		b$, then $\mappingTuple(b) = \mappingTuple(a_{old})$.
	$\mappingInst' = \{ \mappingTuple' \ | \ \mappingTuple \in \mappingInst.\
		\mappingTuple' = \mappingTuple[\symAttrSubSet'] \}$.
\end{definition}



\begin{equation}
	\begin{split}
		\symAttrSubSet'                & = \{ a \ | \  a \in \symAttrSubSet \setminus \symRenameSet_{first} \}  \cup \{ b \ | \ \forall a  \in \symAttrSubSet.\ \exists (a_{old}, b) \in \symRenameSet.\ a = a_{old}\} \\
		\mappingTuple[\symAttrSubSet'] & = \left\{
		\begin{array}{rcl}
			\mappingTuple(a)       & \mbox{for} & \forall a \in \symAttrSubSet \setminus \symRenameSet_{first}                                \\
			\mappingTuple(a_{old}) & \mbox{for} & \forall a_{new} \in \symAttrSubSet'.\ \exists (a_{old}, b) \in \symRenameSet.\  a_{new} = b
		\end{array}
		\right.                                                                                                                                                                                                        \\
		\mappingInst'                  & = \{\mappingTuple[\symAttrSubSet'] \ | \ \mappingTuple \in \mappingInst\}                                                                                                     \\
		\text{Rename}_{R}(\mappingRel) & = (\symAttrSubSet', \mappingInst')
	\end{split}
	\label{eq:rename}
\end{equation}
