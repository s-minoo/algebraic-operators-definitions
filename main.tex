\section{Definitions}%
\label{sec:definitions}


\begin{definition}
	An infinite set $F$ consists of \emph{fragment variables} $f$.
    $F$ is pairwise disjoint with the sets $V$ (SPARQL variables), $I$ (IRIs),
    $B$ (blank nodes), and $L$ (literals).
    There also exists one special fragment variable $f_0 \in F$ that is
    designated as the default fragment variable.
	\label{def:fragments}
\end{definition}


\begin{definition}
	Let $\Omega$ be a multiset of solution mappings, and $P(\Omega)$ be the
	powerset of the multiset $\Omega$.
    A \emph{mapping tuple} $\xi^\Omega$ is a partial function which maps
    \emph{fragment variables} to the sub-multisets of the multiset of solution
    mapping $\Omega$, and it is defined as: $\xi^{\Omega}: F \rightarrow
    P(\Omega)$.
    Let $\Xi$ be a multiset of mapping tuples which we will use as the notation
    throughout the rest of the definitions.
    Note, from now on, we will use the notation $\omega$, as an element of the
    powerset $P(\Omega)$ with $\omega \in P(\Omega)$, to make the definitions
    of the algebraic operators more accessible to read. 
    Here, $\omega$ is also a multiset of solution mappings.
    Combined with Definition~\ref{def:fragments} on the default fragment
    variable $f_0$, there is always a mapping tuple 
    $\xi^{\Omega}_{0}: f_0 = \omega $ with $\omega \in P(\omega)$.
	\label{def:mapping_tuple}
\end{definition}
