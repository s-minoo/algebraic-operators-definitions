\begin{definition}
    A \textbf{normalized} RML document is defined by undertaking the following steps as described 
    in~\citeneeded
    \begin{enumerate}
        \item All shortcuts for constant-term maps have been expanded,
        \item All \texttt{rr:class} definitions are replaced by predicate object maps,
        \item All predicate-object maps with multiple predicate or object
            definitions are expanded into predicate-object maps with a single
            predicate and a single object definition,
        \item All triple maps with multiple predicate-object maps have been
            replaced by a set of equivalent triple maps.
    \end{enumerate}
\end{definition}

	\end{enumerate}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific parent triples map graph
		$\triplesMapGraph{PTM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{PTM}{\triplesMapIRI}$  contain every triple $(s,p,o) \in \symRMLGraph$
		      such that $p$ is the IRI \texttt{rr:parentTriplesMap} and $o = \triplesMapIRI$.
		\item If $(s,\texttt{rr:parentTriplesMap},o) \subseteq \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s'=s$ and  $p'$ is the IRI \texttt{rr:joinCondition}.
		\item If $(s,\texttt{rr:joinCondition},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s' = o$ and  $p'$ is the IRI \texttt{rr:parent}.
	\end{enumerate}
\end{definition}
\begin{algorithm}[thbp]
	\caption{QueryInTriplesMap: Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
	\begin{algorithmic}[1]
		\Require  $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ the subject IRI of a triples map
		\Ensure  $\symDataAcc$ a set of query expressions
		\State $\symDataAcc \gets \emptyset$ \Comment Initial empty set of query expressions
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rml:reference}$ and  $o\in \symAllLiterals$}
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rr:template}$ and  $o\in \symAllLiterals$}
		    \State $\symDataAcc' \gets$ set of all substrings in $o$ that are enclosed by \{ and \}
		    \State $\symDataAcc \gets \symDataAcc \cup \symDataAcc'$
		\EndFor
        \State $T_{child} \gets \{ (s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI} \, | \, p = \texttt{rr:child} \}$
        \State $T_{parent} \gets \{(s,p,o) \in \triplesMapGraph{PTM}{\triplesMapIRI} \, | \, p = \texttt{rr:parent} \}$
        \For{$(s,p,o) \in T_{child} \cup T_{parent}$ such that $o \in \symAllLiterals$ }
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\State \Return $\symDataAcc$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{TranslateSourceOperator: Source operator translation for a given triples map IRI $\triplesMapIRI$}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the source operator for
		\Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
        \State $ls_{iri} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = \triplesMapIRI$  and $p= \texttt{rml:logicalSource}$
        \State $\queryExpr_{root} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = ls_{iri}$  and $p=  \texttt{rml:iterator }$
        \State $\symDataAcc \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$} \Comment{Algorithm~\ref{alg:query_extraction}}
        \State $\symAttrQueryMap \gets$ initially empty function from attributes to query expressions in $\symDataAcc$
        \For{$\queryExpr \in \symDataAcc$}
          \State $\attr \gets$ a fresh attribute from $\symAttrUniverse \setminus \{ \attr_\textrm{s}, \attr_\textrm{p}, \attr_\textrm{o}, \attr_\textrm{g} \}$ that has not been used before in this translation process
          \State $\symAttrQueryMap \gets \symAttrQueryMap \cup \{ a \rightarrow \queryExpr \}$  
        \EndFor
		\State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$ 
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Projection operator translation}
	\label{alg:projection_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the projection operator for
		\Ensure $\symProjSet$ a set of projection attributes
		\Procedure{TranslateProjectionOperator}{$\triplesMapIRI,\symRMLGraph$}
		\State $\symAttrQueryMap \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$}
		\State $\symProjSet \gets \fctDom{\symAttrQueryMap}$
		\State \Return  $\symProjSet$
		\EndProcedure
	\end{algorithmic}
\end{algorithm}
