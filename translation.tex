\section{RML translation}%
\label{sec:rml_translation}


\begin{algorithm}[thbp]
	\caption{Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
    \begin{algorithmic}[1]
        \Require  $\symRMLGraph$ an RDF graph of RML document, 
        $tm_{iri}$ the subject IRI of a triples map
        \Ensure  $\symAttrQueryMap$ a map of attributes to query expressions
        \Procedure{QueryInTriplesMap}{$tm_{iri}, \symRMLGraph$}
            \State $\symAttrQueryMap \gets$ hash map with type: $\symAttrUniverse \setminus \{\fragAttr\} \rightarrow \symDataAcc$ 
            \State $\symAttrQueryMap \gets$ \Call{QueryInReferences}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State $\symAttrQueryMap \gets$ \Call{QueryInJoinConditions}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State $\symAttrQueryMap \gets$ \Call{QueryInTemplate}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State \Return $\symAttrQueryMap$
        \EndProcedure
        \Statex
        \Function{QueryInReferences}{$tm_{iri}, \symRMLGraph, \symAttrQueryMap$}
            \State $p_{ref} \gets$ rml:reference
            \For{$trip \in$ triples with predicate $p_{temp}$, associated with $tm_{iri}$ in $\symRMLGraph$}
                \State $\queryExpr \gets trip.o$\Comment{object literal is a query expression}
                \State $\attr \gets$ unique attribute derived based on $\queryExpr$
                \State $\symAttrQueryMap[\attr] \gets \queryExpr$
            \EndFor
            \State \Return $\symAttrQueryMap$
        \EndFunction
        \Statex
        \Function{QueryInJoinConditions}{$tm_{iri}, \symRMLGraph, \symAttrQueryMap$}
            \State $p_{child} \gets$ rr:child
            \State $p_{parent} \gets$ rr:parent
            \State $T_{child} \gets$  triples with predicate $p_{child}$, where $tm_{iri}$ is the child triples map in $\symRMLGraph$
            \State $T_{parent} \gets$  triples with predicate $p_{parent}$, where $tm_{iri}$ is the parent triples map in $\symRMLGraph$

            \For{$trip \in T_{child} \cup T_{parent}$ }
                \State $\queryExpr \gets trip.o$\Comment{object literal is a query expression}
                \State $\attr \gets$ unique attribute derived based on $\queryExpr$
                \State $\symAttrQueryMap[\attr] \gets \queryExpr$
            \EndFor
            \State \Return $\symAttrQueryMap$
        \EndFunction
        \Statex
        \Function{QueryInTemplate}{$tm_{iri}, \symRMLGraph, \symAttrQueryMap$}
        \State $p_{temp} \gets$ rr:template
            \For{$trip \in$ triples with predicate $p_{temp}$, associated with $tm_{iri}$ in $\symRMLGraph$}
            \State $lit_{temp} \gets trip.o$\Comment{object literal with template string}
                \For{$\queryExpr \in$ query expressions surrounded by ``\{ \}" in $lit_{temp}$}
                    \State $\attr \gets$ unique attribute derived based on $\queryExpr$
                    \State $\symAttrQueryMap[\attr] \gets \queryExpr$
                \EndFor
            \EndFor
        \State \Return $\symAttrQueryMap$
        \EndFunction
    \end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{Source operator translation}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
        \Require $\symRMLGraph$ an RDF graph of RML document, 
        $tm_{iri}$ subject IRI of the triples map to translate the source operator for
        \Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
        \Procedure{TranslateSourceOperator}{$tm_{iri},\symRMLGraph$}
            \State $p_{ls} \gets $ rml:logicalSource
            \State $p_{iter} \gets $ rml:iterator
            \State $ls_{iri} \gets $ object IRI of the triple with the subject~$tm_{iri}$ and the predicate~$p_{ls}$ 
            \State $\queryExpr_{root} \gets $ object literal of the triple with subject $ls_{iri}$ and predicate~$p_{iter}$
            \State $\symAttrQueryMap \gets$ \Call{QueryInTriplesMap}{$tm_{iri}, \symRMLGraph$}
            \State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$
        \EndProcedure
	\end{algorithmic}
\end{algorithm}
