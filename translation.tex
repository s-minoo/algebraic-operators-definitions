\subsection{Translation preliminaries}%
\label{sub:translation_preliminaries}


\todoi{Mariano's paper (Section 5) introduces steps to \textbf{normalize} RML document. We can reference it.}
%https://www.sciencedirect.com/science/article/pii/S1570826815000153?via=ihub 
\begin{definition}
	A \textbf{normalized} RML document is defined by undertaking the following steps as described
	in~\citeneeded
	\begin{enumerate}
		\item All shortcuts for constant-term maps have been expanded,
		\item All \texttt{rr:class} definitions are replaced by predicate object maps,
		\item All predicate-object maps with multiple predicate or object
		      definitions are expanded into predicate-object maps with a single
		      predicate and a single object definition,
		\item All triple maps with multiple predicate-object maps have been
		      replaced by a set of equivalent triple maps.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	\todo{Use $u$ instead of $\mu$.}
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific triples map graph
		$\triplesMapGraph{TM}{\triplesMapIRI} \subseteq \symRMLGraph$} is an RDF graph
	recursively defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{TM}{\triplesMapIRI}$ contains every triple
		      $(s,p,o) \in \symRMLGraph$ for which $s = \triplesMapIRI$
		\item if $(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{TM}{\triplesMapIRI}$ also contains every triple
		      $(s',p',o') \in \symRMLGraph$ where $s' = o$.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific parent triples map graph
		$\triplesMapGraph{PTM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{PTM}{\triplesMapIRI}$  contain every triple $(s,p,o) \in \symRMLGraph$
		      such that $p$ is the IRI \texttt{rr:parentTriplesMap} and $o = \triplesMapIRI$.
		\item If $(s,\texttt{rr:parentTriplesMap},o) \subseteq \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s'=s$ and  $p'$ is the IRI \texttt{rr:joinCondition}.
		\item If $(s,\texttt{rr:joinCondition},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s' = o$ and  $p'$ is the IRI \texttt{rr:parent}.
	\end{enumerate}
\end{definition}


\subsection{Built-in extend functions}%
\label{sub:built_in_extend_functions}

The following definitions describe the provided built-in extend functions used
when translating RML documents to algebraic mapping expressions as described in
Section~\ref{sub:translation_algorithms}.

\subsubsection{ToIRI function}%
\label{ssub:toiri_function}
\begin{definition}
	Let $v \in \symTermUniverse \cup \error$, be the input value.
	The extend function $\toIRI: (\symTermUniverse \cup \error) \rightarrow (\symTermUniverse \cup \error)$
	is a unary function, and we denote $\fctToIRI{v}$ as a function 
    that converts string literals representing IRIs into these IRIs. 
    Formally,  for every $v \in (\symTermUniverse \cup \error)$: 
    \begin{equation*}
    	\fctToIRI{v} =
    	\begin{cases}
    		\mathit{lex}
    		& \text{if $v$ is a literal $\literalTuple$ such that $\lex$ is a valid IRI}
    		\\[-2mm]
    		& \text{and $\dt$ is the IRI \texttt{xsd:string},}
    		\\
    		\error
    		& \text{else.}
    	\end{cases}
    \end{equation*}
\end{definition}



\subsubsection{ToBNode function}%
\label{ssub:tobnode_function}
\begin{definition}
	Let $L2B: \symAllLiterals \rightarrow \symAllBNodes$ a global state on the
	mapping of literals to blank nodes, and $v\in \symTermUniverse \cup \error$
	an input literal value.
	The extend function $\toBNode: (\symTermUniverse \cup \error) \rightarrow (\symAllBNodes \cup \error)$
	is a unary function, and we denote $\fctToBNode{v}$ as type casting of the input literal value
	$v$ to a blank node as follows:
	\begin{equation*}
		\fctToBNode{v} =
		\begin{cases}
			L2B(v), & \text{if $v \in \fctDom{L2B}$ }                                   \\
			v',     & \text{ if $v'$ is a blank node with label $v$ and $L2B(v) = v'$ } \\
			\error, & \text{else}
		\end{cases}
	\end{equation*}
    $\fctToBNode{v}$ also updates the global state $L2B$ with 
    $L2B \cup \{(v \rightarrow v')\}$  if the generated blank node $v' \notin \fctImg{L2B}$. 
\end{definition}

{\color{blue}
We assume a function $\mathit{L2B}: \symAllLiterals \rightarrow \symAllBNodes$ that is injective.
$\toBNode$ is a unary extend function that uses $\mathit{L2B}$ to convert string literals into blank nodes. Formally, for every $v \in (\symTermUniverse \cup \error)$,
\begin{equation*}
	\toBNode{v} =
	\begin{cases}
		\mathit{L2B}(\mathit{lex})
		& \text{if $v$ is a literal $(\mathit{lex}, \mathit{dt})$ such that $\mathit{dt}$ is the IRI \texttt{xsd:string},}
		\\
		\error
		& \text{else.}
	\end{cases}
\end{equation*}
}



\subsubsection{ToLiteral function}%
\label{ssub:toliteral_function}


\begin{definition}
    Let $dt$ be an optional RDF literal data type, $lt$ be an optional RDF
    literal language tag, and $v\in \symTermUniverse \cup \error$ an input value.
	The extend function $\toLiteral: (\symTermUniverse \cup \error) \rightarrow (\symAllLiterals  \cup \error)$
    is a unary function, and we denote $\fctToLiteral{v}$ as type casting of
    the input value $v$ to an RDF literal node as follows:

    \begin{equation*}
        \fctToLiteral{v} = 
        \begin{cases}
            v', &\parbox{15em}{ if $dt$ and $lt$ are not provided, $v'$ is literal with value as lexical form of $v$ and datatype \texttt{xsd:string}   } \\  \\ 
            v', &\parbox{15em}{ if $dt$ is provided, $v'$ is literal with value as lexical form of $v$ and datatype $dt$   } \\  \\ 
            v', &\parbox{15em}{ if $lt$ is provided, $v'$ is literal with value as lexical form of $v$ and language tag $lt$   } \\ \\ 
            \error, & \text{ else }
        \end{cases}
    \end{equation*}
\end{definition}




\subsubsection{Concatenate function}%
\label{ssub:concatenate_function}

\begin{definition}
Let $v,w \in \symTermUniverse \cup \error$ two input values. 
The extend function $\concat: (\symAllLiterals \cup \error) \times  (\symAllLiterals \cup \error) \rightarrow  (\symAllLiterals \cup \error) $
    is a binary function, and we denote $\fctConcat{v}{w}$ as a literal 
$v'$ with its value as the concatenation, $v \ || \ w$,  
of the lexical forms of the two input values $v$ and $w$. 
For a sequence of input values $\bar{V} = (v_1,\dots,v_n)$ with $n \ge 2$, we 
denote $\fctConcatSeq{V}$  as the application of 
$\concat$ functions on the sequence as such 
$\fctConcat{v_1}{\dots, \fctConcat{v_{n-1}}{v_n}}$
\end{definition}


\subsubsection{Template function}%
\label{ssub:template_function}
\begin{definition}
	Let $v\in \symTermUniverse \cup \error$ an input value,
	$\symAttrQueryMap \!: \symAttrUniverse  \rightarrow \symDataAcc$ a partial
	function.
	The extend function $\template: (\symAllLiterals \cup \error) \rightarrow  \extExpr$
	is a unary function, and we denote  $\fctTemplate{v}$
    as the extend expression~$\extExpr$ tuple $(concatSeq, \extExpr_1, \dots, \extExpr_n)$ 
    such that for each $s_n \in \symAllStrings$, where $s_n$ is the $n$-th substring in $v$,
    the value for $\extExpr_n$ is determined as follows: 
    if $s_n$ is enclosed by ``\{" and ``\}", it is a string representation of 
    the query expression~$\queryExpr \in \symDataAcc$,
    and the corresponding $\extExpr_n = \symAttrQueryMap^{-1}(q)$, 
    otherwise $\extExpr_n$ is an RDF literal with the value of $s_n$.
\end{definition}

\begin{algorithm}[htbp]
	\caption{Template: An extend function which resolves the template string of
		the input literal node and returns the template resolved string as a
		literal}
	\label{alg:template_function}
	\begin{algorithmic}[1]
		\Require $v$ an input literal value, $\symAttrQueryMap : \symAttrUniverse \rightarrow \symDataAcc$
		a mapping of attribute to query expression
		\Ensure $v'$ an RDF literal node or an error
		\State $S \gets $ a sequence of substrings, $s$, of $v$ where substring $s$
		appears in the order that it appears in $v$ and $s$ is either enclosed by
		unescaped pairs of ``\{" and ``\}" or not
		\State $v' \gets $ empty string literal
		\For {$s \in S$}
		\State $s' \gets s$
		\If{$s$ is enclosed by paris of ``\{" and ``\}" and $s \in \fctImg{\symAttrQueryMap}$}
		\State $s' \gets \symAttrQueryMap^{-1}(s) $
		\Else
		\State \Return $\error$
		\EndIf
		\State $v' \gets v' \ || \  s'$
		\EndFor
		\State \Return $v'$
	\end{algorithmic}
\end{algorithm}

\subsection{Translation algorithms}%
\label{sub:translation_algorithms}




\begin{algorithm}[thbp]
	\caption{QueryInTriplesMap: Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
	\begin{algorithmic}[1]
		\Require  $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ the subject IRI of a triples map
		\Ensure  $\symDataAcc$ a set of query expressions
		\State $\symDataAcc \gets \emptyset$ \Comment Initial empty set of query expressions
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rml:reference}$ and  $o\in \symAllLiterals$}
		\State $\queryExpr \gets $ lexical form of $o$
		\State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rr:template}$ and  $o\in \symAllLiterals$}
		\State $\symDataAcc' \gets$ set of all substrings in $o$ that are enclosed by \{ and \}
		\State $\symDataAcc \gets \symDataAcc \cup \symDataAcc'$
		\EndFor
		\State $T_{child} \gets \{ (s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI} \, | \, p = \texttt{rr:child} \}$
		\State $T_{parent} \gets \{(s,p,o) \in \triplesMapGraph{PTM}{\triplesMapIRI} \, | \, p = \texttt{rr:parent} \}$
		\For{$(s,p,o) \in T_{child} \cup T_{parent}$ such that $o \in \symAllLiterals$ }
		\State $\queryExpr \gets $ lexical form of $o$
		\State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\State \Return $\symDataAcc$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}[thbp]
	\caption{ExtractExtendOp: Extend operator translation from an RML term map}
	\label{alg:rml_termmap_extend_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
        $L2B$ a global state mapping literals to blank nodes, 
		$\symAttrQueryMap$ a partial function mapping attributes to query expression,
		$tm$ subject IRI of the RML term map to translate the extend operator for
		\Ensure $\extExpr$ an extend expression
        \State $\extExpr \gets $ initially empty extend expression

		\If{$(tm, \texttt{rr:reference}, o) \in \triplesMapGraph{TM}{tm}$}
    		\State $\extExpr \gets a$, where $a \in \fctDom{\symAttrQueryMap}$ and $\symAttrQueryMap(a)$ is the lexical form of $o$
		\ElsIf{$(tm, \texttt{rr:constant}, o) \in \triplesMapGraph{TM}{tm}$}
    		\State $\extExpr \gets o$
		\ElsIf{$(tm, \texttt{rr:template}, o) \in \triplesMapGraph{TM}{tm}$}
            \State  $\extExpr \gets $ \Call{Template}{$o, \symAttrQueryMap$}  \Comment Algorithm~\ref{alg:template_function}
		\EndIf

        \State TODO remove:  $tt \gets o_{tt} $ in triple $(tm,\texttt{rr:termType},o_{tt}) \in \triplesMapGraph{TM}{tm}$ \Comment extract term type
        \If{there exists a triple $(tm,\texttt{rr:termType}, \texttt{rr:BlankNode}) \in \triplesMapGraph{TM}{tm}$}
            \State $\extExpr \gets $\Call{toBNode}{$\extExpr, L2B$}
        \ElsIf{$tt = \texttt{rr:BlankNode}$}
            \State $\extExpr \gets $\Call{toIRI}{$\extExpr$}
        \Else
            \State $dt \gets \texttt{xsd:string}$ 
            \State $lt \gets $ undefined
		    \If{$(tm, \texttt{rr:languageMap}, o) \in \triplesMapGraph{TM}{tm}$}
                \State $lt \gets$ \Call{ExtractExtendOp}{$\symRMLGraph, L2B, \symAttrQueryMap, o$} \Comment RML language map is a term map 
		    \ElsIf{$(tm, \texttt{rr:language}, o) \in \triplesMapGraph{TM}{tm}$}
                \State $lt \gets o$ with $o$ a literal node that specifies the language tag
		    \ElsIf{there exists a triple $(tm, \texttt{rr:datatype}, o) \in \triplesMapGraph{TM}{tm}$ such that $o \in \symAllIRIs$}
    		    \State $dt \gets o$
            \EndIf
            \State $\extExpr \gets $\Call{toLiteral}{$\extExpr, dt, lt$}
		\EndIf
		\State \Return $\extExpr$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}[thbp]
	\caption{TranslateSourceOperator: Source operator translation for a given triples map IRI $\triplesMapIRI$}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the source operator for
		\Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
		\State $ls_{iri} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = \triplesMapIRI$  and $p= \texttt{rml:logicalSource}$
		\State $\queryExpr_{root} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = ls_{iri}$  and $p=  \texttt{rml:iterator }$
		\State $\symDataAcc \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$} \Comment{Algorithm~\ref{alg:query_extraction}}
		\State $\symAttrQueryMap \gets$ initially empty function from attributes to query expressions in $\symDataAcc$
		\For{$\queryExpr \in \symDataAcc$}
		\State $\attr \gets$ a fresh attribute from $\symAttrUniverse \setminus \{ \attr_\textrm{s}, \attr_\textrm{p}, \attr_\textrm{o}, \attr_\textrm{g} \}$ that has not been used before in this translation process
		\State $\symAttrQueryMap \gets \symAttrQueryMap \cup \{ a \rightarrow \queryExpr \}$
		\EndFor
		\State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$
	\end{algorithmic}
\end{algorithm}

\begin{definition}
    \label{def:mapping_definition}
    A \textbf{mapping definition}  is an expression that can be built by 
    combining various operators of the mapping algebra. 
\end{definition}

\begin{definition}
    \label{def:init_data_source}
    Let $ls_{iri}$ be an IRI of an RML logical source. 
    We denote $\fctInitSrc{ls_{iri}}$ as a \textbf{data source}, $\dataSourceTuple$, derived 
    from the configurations of the given logical source~$ls_{iri}$.
\end{definition}
For example, given an RDF graph~$\symRMLGraph$ of an RML document in Listing~\ref{lst:rml_document}, 
and a triple $(s, \texttt{rml:logicalSource}, ls_{iri}) \in \symRMLGraph$, 
$\fctInitSrc{ls_{iri}}$ returns a data source tuple, $\dataSourceTuple$, such 
that $\symDataset$ is a CSV dataset from the file ``sport.csv" and 
$\sourceType =  \sourceTypeTuple$ such that  $\symSetDataset$ is a set of  
CSV datasets, $\symDataAcc$ a set of all possible query expressions applicable to the 
given CSV dataset, such as column names.


\begin{algorithm}[htbp]
	\caption{RML translation to algebraic mapping expressions}
	\label{alg:rml2algebra}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of a \textbf{normalized} RML document
        \State $\omega \gets$ initially empty mapping definition \Comment Definition~\ref{def:mapping_definition}
		%% TODO Define earlier: a mapping definition is an expression that can be built by combining the various operators of our algebra
		\For{$(\triplesMapIRI, \texttt{rdf:type}, \texttt{rr:TriplesMap}) \in \symRMLGraph$}
	    	\State  $\iterConfigTuple \gets$ \Call{TranslateSourceOperator}{$\symRMLGraph, \triplesMapIRI$} \Comment{Algorithm~\ref{alg:rml_source_op}}
            \State   $src \gets \textrm{initSrc}(o')$ for $(\triplesMapIRI,\texttt{rr:logicalSource},o') \in \triplesMapGraph{TM}{\triplesMapIRI}$ \Comment Definition~\ref{def:init_data_source}
	    	\State   $\omega \gets \fctSourceOp{\iterConfigTuple}{src}$
	    	\State  $(v_{subj}, v_{pred},v_{obj}) \gets $ initial placeholder value for the (s,p,o) attributes for RDF serialization

		\For{$(s,\texttt{rr:subjectMap},o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $s = \triplesMapIRI$}
            \State $\extExpr \gets $\Call{ExtractExtendOp}{$\symRMLGraph, L2B, \symAttrQueryMap, o$} \Comment Algorithm~\ref{alg:rml_termmap_extend_op}
    		\State $\omega \gets \extOp(\omega)$
		\EndFor
		\For{$(s,\texttt{rr:predicateMap},o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ }
            \State $\extExpr \gets $\Call{ExtractExtendOp}{$\symRMLGraph, L2B, \symAttrQueryMap, o$}
    		\State $\omega \gets \extOp(\omega)$
		\EndFor \Comment Continued in Algorithm~\ref{alg:rml2algebra_part2}
		\State\dots
		\algstore{state:full_translation}
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
	\caption{RML translation to algebraic mapping expressions}
	\label{alg:rml2algebra_part2}
	\begin{algorithmic}[1]
		\algrestore{state:full_translation}
		\State \dots
            \For{$(s,\texttt{rr:objectMap},o) \in \triplesMapGraph{TM}{\triplesMapIRI}$}
                \If{$(o,\texttt{rr:parentTriplesMap},o') \in \triplesMapGraph{TM}{\triplesMapIRI}$}
		            \State $s_{ptm} \gets o'$
		            \State  $\iterConfig \gets $ \Call{TranslateSourceOperator}{$\symRMLGraph, s_{ptm}$} \Comment{Algorithm~\ref{alg:rml_source_op}}
                    \State   $source_{ptm} \gets \fctInitSrc{o''}$ with $(s'',\texttt{rr:logicalSource},o'') \in \triplesMapGraph{TM}{s_{ptm}}$ 
		            \State  $\omega_{ptm} \gets \fctSourceOp{\iterConfig}{source_{ptm}}$
                    \If{$(o,\texttt{rr:joinCondition},o') \in \triplesMapGraph{TM}{\triplesMapIRI}$}
                        \State  $a \gets \symAttrQueryMap^{-1}(o'') $ for $o''$ in triple $(o',\texttt{rr:child},o'') \in \triplesMapGraph{TM}{\triplesMapIRI}$
                        \State  $a_{ptm} \gets \symAttrQueryMap^{-1}(o'')$ for $o''$ in triple $(o',\texttt{rr:parent},o'') \in \triplesMapGraph{TM}{\triplesMapIRI}$
		                \State  $\symJoinAttrPairs \gets \{ (a, a_{ptm}) \}$
		                \State  $\omega \gets \fctEquiJoinOp{\omega}{\omega_{ptm}}$
                    \Else
		                \State  $\omega \gets $ TODO: union join
                    \EndIf
		            \State $\extOp \gets$ extend operator for the subject map of triples map with IRI $s_{ptm}$
		            \State  $v_{obj} \gets a$  with $a$ the attribute used by $\extOp$
		            \State  $\omega \gets \fctExtOp{\omega}$
		            \State  \texttt{break}
		        \EndIf
		        \State $(a, \extExpr) \gets $ extend operator from triple $(o,\texttt{rr:template},o') \lor (o,\texttt{rr:reference},o')$
		        \State $\extExpr \gets $\Call{TypeCast}{$\extExpr$} according to \texttt{rr:termType}
		        \State  $v_{obj} \gets a$
		        \State $\omega \gets \extOp(\omega)$
		    \EndFor
		    \State $\omega \gets $ \Call{SerializeOp}{$(v_{subj}, v_{pred},v_{obj}, \omega)$}
		    \State $Execute(\omega)$
		    \State $\omega \gets$  initialization value for the current algebraic mapping expressions
		\EndFor
	\end{algorithmic}
\end{algorithm}
