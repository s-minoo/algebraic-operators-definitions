\subsection{Translation Preliminaries}%
\label{sub:translation_preliminaries}


\todoi{Mariano's paper (Section 5) introduces steps to \textbf{normalize} RML document. We can reference it.}
%https://www.sciencedirect.com/science/article/pii/S1570826815000153?via=ihub 
\begin{definition}
	A \textbf{normalized} RML document is defined by undertaking the following steps as described
	in~\citeneeded
	\begin{enumerate}
		\item All shortcuts for constant-term maps have been expanded,
		\item All \texttt{rr:class} definitions are replaced by predicate object maps,
		\item All predicate-object maps with multiple predicate or object
		      definitions are expanded into predicate-object maps with a single
		      predicate and a single object definition,
		\item All triple maps with multiple predicate-object maps have been
		      replaced by a set of equivalent triple maps.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific triples map graph
		$\triplesMapGraph{TM}{\triplesMapIRI} \subseteq \symRMLGraph$} is an RDF graph
	recursively defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{TM}{\triplesMapIRI}$ contains every triple
		      $(s,p,o) \in \symRMLGraph$ for which $s = \triplesMapIRI$
		\item if $(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{TM}{\triplesMapIRI}$ also contains every triple
		      $(s',p',o') \in \symRMLGraph$ where $s' = o$.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific parent triples map graph
		$\triplesMapGraph{PTM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{PTM}{\triplesMapIRI}$  contain every triple $(s,p,o) \in \symRMLGraph$
		      such that $p$ is the IRI \texttt{rr:parentTriplesMap} and $o = \triplesMapIRI$.
		\item If $(s,\texttt{rr:parentTriplesMap},o) \subseteq \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s'=s$ and  $p'$ is the IRI \texttt{rr:joinCondition}.
		\item If $(s,\texttt{rr:joinCondition},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains every triple $(s',p',o') \in \symRMLGraph$
		      such that $s' = o$ and  $p'$ is the IRI \texttt{rr:parent}.
	\end{enumerate}
\end{definition}


\subsection{Built-in Extend Functions}%
\label{sub:built_in_extend_functions}

The following definitions describe the provided built-in extend functions used
when translating RML documents to algebraic mapping expressions as described in
Section~\ref{sub:translation_algorithms}.

\subsubsection{toIRI function}%
\label{ssub:toiri_function}
$\toIRI$
is a unary extension function that converts string literals representing IRIs into these IRIs. 
Formally,  for every $v \in (\symTermUniverse \cup \error)$: 
\begin{equation*}
	\fctToIRI{v} =
	\begin{cases}
		\mathit{lex}
		& \text{if $v$ is a literal $\literalTuple$ such that $\lex$ is a valid IRI}
		\\[-2mm]
		& \text{and $\dt$ is the IRI \texttt{xsd:string},}
		\\
		\error
		& \text{else.}
	\end{cases}
\end{equation*}



\subsubsection{toBNode function}%
\label{ssub:tobnode_function}
We assume a function $\LTB: \symAllLiterals \rightarrow \symAllBNodes$ that is injective.
$\toBNode$ is a unary extension function that uses $\LTB$ to convert string literals into blank nodes. Formally, for every $v \in (\symTermUniverse \cup \error)$:
\begin{equation*}
	\fctToBNode{v} =
	\begin{cases}
		\LTB(\lex)
		& \text{if $v$ is a literal $\literalTuple$ such that $\dt$ is the IRI \texttt{xsd:string},}
        \\ 
		\error
		& \text{else.}
	\end{cases}
\end{equation*}


\subsubsection{toLiteral function}%
\label{ssub:toliteral_function}
$\toLiteral$
is a binary extension function that converts string literals
into literals with the 
datatype IRI
that is given as the second argument.
Formally, for every pair $(v, \dt)\in (\symTermUniverse \cup \error) \times (\symTermUniverse \cup \error)$:

\begin{equation*}
    \fctToLiteral{v, \dt} = 
    \begin{cases}
        (\lex, \dt) &\text{if $v$ is a literal~$(\lex, \dt')$ such that $\dt'$ is the IRI \texttt{xsd:string}}, \\
        \error & \text{else.}
    \end{cases}
\end{equation*}




\subsubsection{Concatenate function}%
\label{ssub:concatenate_function}
$\concat$ is a binary extension function that concatenates two string literals.
Formally, for every pair $(v_1, v_2) \in (\symTermUniverse \cup \error) \times (\symTermUniverse \cup \error)$

\begin{equation*}
    \fctConcat{v_1}{v_2} = 
    \begin{cases}
        (\lex_1 \circ \lex_2, \texttt{xsd:string})  &\text{if $v_1$ and $v_2$ are literals $(\lex_1,\dt_1)$ and}
		\\[-1mm]
        & \text{$(\lex_2,\dt_2)$, respectively, such that both}
		\\[-1mm]
        & \text{$\dt_1$ and $\dt_2$ are the IRI \texttt{xsd:string},}
        \\ 
        \error &\text{else,}
		\\
    \end{cases}
\end{equation*}
where $\lex_1 \circ \lex_2$ is the string obtained from the concatenating the strings $\lex_1$ and $\lex_2$.

\subsection{Translation Algorithms}%
\label{sub:translation_algorithms}

\begin{algorithm}[htbp]
	\caption{QueryInTriplesMap: Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
	\begin{algorithmic}[1]
		\Require  $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ the subject IRI of a triples map
		\Ensure  $\symDataAcc$ a set of query expressions
		\State $\symDataAcc \gets \emptyset$ \Comment Initial empty set of query expressions
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rml:reference}$ and  $o\in \symAllLiterals$}
            \State $\queryExpr \gets \lex$ where $o = (\lex,\dt)$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rr:template}$ and  $o\in \symAllLiterals$}
		    \State $\symDataAcc' \gets$ set of all substrings in $o$ that are enclosed by \{ and \}
		    \State $\symDataAcc \gets \symDataAcc \cup \symDataAcc'$
		\EndFor
		\State $T_{child} \gets \{ (s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI} \, | \, p = \texttt{rr:child} \}$
		\State $T_{parent} \gets \{(s,p,o) \in \triplesMapGraph{PTM}{\triplesMapIRI} \, | \, p = \texttt{rr:parent} \}$
		\For{$(s,p,o) \in T_{child} \cup T_{parent}$ such that $o \in \symAllLiterals$ }
            \State $\queryExpr \gets \lex$ where $o = (\lex,\dt)$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\State \Return $\symDataAcc$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}[htbp]
	\caption{ExtractExtendExpr: Extend expression translation from an RML term map}
	\label{alg:rml_termmap_extend_expr}
	\begin{algorithmic}[1]
		\Require 
             $\symRMLGraph$ an RDF graph of RML document,
            \Statex \qquad \quad $\LTB$ an injective function mapping literals to blank nodes, 
            \Statex \qquad \quad $\symAttrQueryMap$ a partial function mapping attributes to query expression,
            \Statex \qquad \quad $tm$ subject IRI of the RML term map to translate the extend operator for
		\Ensure $\extExpr$ an extend expression
        \State $\extExpr \gets $ initially empty extend expression

        \If{$(tm, \texttt{rr:reference}, o) \in \triplesMapGraph{TM}{tm}$ such that $o$ is a literal $\literalTuple$ }
    		\State $\extExpr \gets a$, where $a$ is the attribute in $\fctDom{\symAttrQueryMap}$ such that $\symAttrQueryMap(a) = \lex$
		\ElsIf{$(tm, \texttt{rr:constant}, o) \in \triplesMapGraph{TM}{tm}$}
    		\State $\extExpr \gets o$
		\ElsIf{$(tm, \texttt{rr:template}, o) \in \triplesMapGraph{TM}{tm}$ such that $o$ is a literal}
            \State  $\extExpr \gets $ \Call{Template}{$o, \symAttrQueryMap$}  \Comment Algorithm~\ref{alg:template_function}
		\EndIf

        \If{$(tm,\texttt{rr:termType}, \texttt{rr:BlankNode}) \in \triplesMapGraph{TM}{tm}$}
            \State $\extExpr \gets \fctToBNode{\extExpr}$
        \ElsIf{$(tm,\texttt{rr:termType}, \texttt{rr:IRI}) \in \triplesMapGraph{TM}{tm}$}
            \State $\extExpr \gets \fctToIRI{\extExpr}$

        \ElsIf{$(tm, \texttt{rr:datatype}, o) \in \triplesMapGraph{TM}{tm}$ such that $o \in \symAllIRIs$}
            \State $\extExpr \gets \fctToLiteral{\extExpr, o}$
        \Else 
            \State $\extExpr \gets \fctToLiteral{\extExpr, \texttt{xsd:string}}$
		\EndIf
		\State \Return $\extExpr$
	\end{algorithmic}
\end{algorithm}

\begin{definition}
    \label{def:tempSubStrs}
    Let $v = (\lex, \dt)$ an input literal string such that $\dt$ is \texttt{xsd:string}.  
    For ease of reading, we denote a substring as \textbf{enclosed query substring} 
    if it is enclosed by a pair of curly braces \{ and \} as defined in R2RML~\citeneeded, 
    otherwise it is denoted as \textbf{normal substring}.
    $\tempSubStrs$ is a unary function which splits the lexical form of the 
    input literal string into a sequence of substrings~$\substrSeq$ which contains 
    alternating normal substrings and enclosed query substrings in two pass as follows: 

    \begin{enumerate}
        \item  An intermediate sequence of substrings~$\substrSeq'$ is generated which 
            contains substrings that are \textbf{normal} or \textbf{enclosed query}  
            substrings in the order that these substrings appear in $\lex$ from 
            left to right. 
        \item $\substrSeq'$ is further processed as follows to produce $\substrSeq$: 
            \begin{enumerate}
                \item if $\substrSeq'$ is empty, an \textbf{empty substring} is inserted
                \item if there are two enclosed query substring \textbf{next to each other} in $\substrSeq'$, 
                    insert an empty substring between them
                \item if the first element in $\substrSeq'$ is an enclosed query substring, 
                    add an empty substring at the head of $\substrSeq'$
                \item if the last element in $\substrSeq'$ is an enclosed query substring, 
                    add an empty substring at the tail of $\substrSeq'$
            \end{enumerate}
    \end{enumerate}
    After the aforementioned processing, the sequence $\substrSeq$ has an 
    \textbf{enclosed query substring} at every odd index and \textbf{normal substring} 
    at every even index, and it is non-empty.
\end{definition}

\begin{algorithm}[htbp]
	\caption{Template: An algorithm which extracts an extend expression for 
        resolving the template string of the input literal node.}
	\label{alg:template_function}
	\begin{algorithmic}[1]
        \Require $(\lex, \dt)$ an input literal value, $\symAttrQueryMap : \symAttrUniverse \rightarrow \symDataAcc$
		a mapping of attribute to query expression
		\Ensure $\extExpr$ an extend expression
        \State $\substrSeq \gets \fctTempSubStrs{\lex}$ \Comment Definition~\ref{def:tempSubStrs}
        \State $\extExpr \gets \fctConcat{``"}{``"}$ \Comment Section~\ref{ssub:concatenate_function}
        \For {$s_i \in  (s_0,\dots,s_n)$ where $ \substrSeq = (s_0,\dots,s_n)$ and $n = |\substrSeq|- 1$ }
            \If{$i$ is even}
                \State $\extExpr \gets \fctConcat{\extExpr}{s_i}$ 
            \Else
                \State $\extExpr \gets \fctConcat{\extExpr}{\symAttrQueryMap^{-1}(s_i)}$
            \EndIf
		\EndFor
		\State \Return $\extExpr$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}[thbp]
	\caption{TranslateSourceOperator: Source operator translation for a given triples map IRI $\triplesMapIRI$}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the source operator for
		\Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
		\State $ls_{iri} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = \triplesMapIRI$  and $p= \texttt{rml:logicalSource}$
		\State $\queryExpr_{root} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = ls_{iri}$  and $p=  \texttt{rml:iterator }$
		\State $\symDataAcc \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$} \Comment{Algorithm~\ref{alg:query_extraction}}
		\State $\symAttrQueryMap \gets$ initially empty function from attributes to query expressions in $\symDataAcc$
		\For{$\queryExpr \in \symDataAcc$}
		\State $\attr \gets$ a fresh attribute from $\symAttrUniverse \setminus \{ \attr_\textrm{s}, \attr_\textrm{p}, \attr_\textrm{o}, \attr_\textrm{g} \}$ that has not been used before in this translation process
		\State $\symAttrQueryMap \gets \symAttrQueryMap \cup \{ a \rightarrow \queryExpr \}$
		\EndFor
		\State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$
	\end{algorithmic}
\end{algorithm}

\begin{definition}
    \label{def:mapping_definition}
    A \textbf{mapping definition}  is an expression that can be built by 
    combining various operators of the mapping algebra. 
\end{definition}

\begin{definition}
    \label{def:init_data_source}
    Let $ls_{iri}$ be an IRI of an RML logical source. 
    We denote $\fctInitSrc{ls_{iri}}$ as a \textbf{data source}, $\dataSourceTuple$, derived 
    from the configurations of the given logical source~$ls_{iri}$.
\end{definition}
For example, given an RDF graph~$\symRMLGraph$ of an RML document in Listing~\ref{lst:rml_document}, 
and a triple $(s, \texttt{rml:logicalSource}, ls_{iri}) \in \symRMLGraph$, 
$\fctInitSrc{ls_{iri}}$ returns a data source tuple, $\dataSourceTuple$, such 
that $\symDataset$ is a CSV dataset from the file ``sport.csv" and 
$\sourceType =  \sourceTypeTuple$ such that  $\symSetDataset$ is a set of  
CSV datasets, $\symDataAcc$ a set of all possible query expressions applicable to the 
given CSV dataset, such as column names.
$\eval$ is a function that returns a row of CSV data object per
iteration, and $\cbeval$ is a function that returns the data object for the
given CSV column name.

\begin{algorithm}[htbp]
	\caption{RML translation to algebraic mapping expressions}
	\label{alg:rml2algebra}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of a \textbf{normalized} RML document, 
        $\LTB$ an injective function mapping literals to blank nodes, 
        \Ensure $\omega$ a mapping definition
        \State $\omega \gets$ initially empty mapping definition \Comment Definition~\ref{def:mapping_definition}
		\For{$(\triplesMapIRI, \texttt{rdf:type}, \texttt{rr:TriplesMap}) \in \symRMLGraph$}
	    	\State  $\iterConfigTuple \gets$ \Call{TranslateSourceOperator}{$\symRMLGraph, \triplesMapIRI$} \Comment{Algorithm~\ref{alg:rml_source_op}}
            \State   $src \gets \textrm{initSrc}(o')$ for $(\triplesMapIRI,\texttt{rr:logicalSource},o') \in \triplesMapGraph{TM}{\triplesMapIRI}$ \Comment Definition~\ref{def:init_data_source}
	    	\State   $\omega' \gets \fctSourceOp{\iterConfigTuple}{src}$

            \Statex
            \State $tm_{sub} \gets o$, where triple $(s,\texttt{rr:subjectMap},o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $s = \triplesMapIRI$
            \State $\extExpr \gets $\Call{ExtractExtendExpr}{$\symRMLGraph, \LTB, \symAttrQueryMap, tm_{sub}$} \Comment Algorithm~\ref{alg:rml_termmap_extend_expr}
            \State $\omega' \gets \extOpAttr{\attr_{s}}(\omega')$
            \Statex
            \State $tm_{pred} \gets o$, where triple $(s,\texttt{rr:predicateMap},o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $s = \triplesMapIRI$
            \State $\extExpr \gets $\Call{ExtractExtendExpr}{$\symRMLGraph, \LTB, \symAttrQueryMap, tm_{pred}$}
            \State $\omega' \gets \extOpAttr{\attr_{p}}(\omega')$
		    \State\dots  \Comment Continued in Algorithm~\ref{alg:rml2algebra_part2}
		    \algstore{state:full_translation}
	\end{algorithmic}
\end{algorithm}
