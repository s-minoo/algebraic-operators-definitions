\section{RML translation}%
\label{sec:rml_translation}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific triples map graph
		$\triplesMapGraph{TM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	recursively defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{TM}{\triplesMapIRI}$ contains every triple
		      $(s,p,o) \in \symRMLGraph$ for which $s = \triplesMapIRI$
		\item if $(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{TM}{\triplesMapIRI}$ also contains every triple
		      $(s',p',o') \in \symRMLGraph$ where $s' = o$.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific parent triples map graph
		$\triplesMapGraph{PTM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{PTM}{\triplesMapIRI}$  contains triples $(s,p,o) \in \symRMLGraph$
		      where $p = \texttt{rr:parentTriplesMap}$ and $o = \triplesMapIRI$.
		\item If $(s,\texttt{rr:parentTriplesMap},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains triples $(s',p',o') \in \symRMLGraph$
		      where $s'=s$ and  $p' = \texttt{rr:joinCondition}$.
		\item If $(s,\texttt{rr:joinCondition},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains triples $(s',p',o') \in \symRMLGraph$
		      where $s' = o$ and  $p' = \texttt{rr:parent}$.
	\end{enumerate}
\end{definition}
\begin{algorithm}[thbp]
	\caption{QueryInTriplesMap: Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
	\begin{algorithmic}[1]
		\Require  $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ the subject IRI of a triples map
		\Ensure  $\symDataAcc$ a map of attributes to query expressions
		\State $\symDataAcc \gets \emptyset$ \Comment Initial empty set of query expressions
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rml:reference}$ and  $o\in \symAllLiterals$}
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rr:template}$ and  $o\in \symAllLiterals$}
		    \For{$\queryExpr \in$ substrings in $o$ that are enclosed by \{ and \} }
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		    \EndFor
		\EndFor
        \State $T_{child} \gets \{ (s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI} \} $ where $p = \texttt{rr:child}$
        \State $T_{parent} \gets \{(s,p,o) \in \triplesMapGraph{PTM}{\triplesMapIRI}\}$ where $p = \texttt{rr:parent}$
        \For{$(s,p,o) \in T_{child} \cup T_{parent}$ such that $o \in \symAllLiterals$ }
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\State \Return $\symDataAcc$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{Source operator translation}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the source operator for
		\Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
		\Procedure{TranslateSourceOperator}{$\triplesMapIRI,\symRMLGraph$}
		\State $p_{ls} \gets $ rml:logicalSource
		\State $p_{iter} \gets $ rml:iterator
		\State $ls_{iri} \gets $ object IRI of the triple with the subject~$\triplesMapIRI$ and the predicate~$p_{ls}$
		\State $\queryExpr_{root} \gets $ object literal of the triple with subject $ls_{iri}$ and predicate~$p_{iter}$
		\State $\symAttrQueryMap \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$}
		\State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$
		\EndProcedure
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{Projection operator translation}
	\label{alg:projection_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the projection operator for
		\Ensure $\symProjSet$ a set of projection attributes
		\Procedure{TranslateProjectionOperator}{$\triplesMapIRI,\symRMLGraph$}
		\State $\symAttrQueryMap \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$}
		\State $\symProjSet \gets \fctDom{\symAttrQueryMap}$
		\State \Return  $\symProjSet$
		\EndProcedure
	\end{algorithmic}
\end{algorithm}
