\section{RML translation}%
\label{sec:rml_translation}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific triples map graph
		$\triplesMapGraph{TM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	recursively defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{TM}{\triplesMapIRI}$ contains every triple
		      $(s,p,o) \in \symRMLGraph$ for which $s = \triplesMapIRI$
		\item if $(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{TM}{\triplesMapIRI}$ also contains every triple
		      $(s',p',o') \in \symRMLGraph$ where $s' = o$.
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\symRMLGraph$ be an RDF graph of an RML mapping document, and
	$\triplesMapIRI$ an IRI of a RML triples map.
	A \textbf{$\triplesMapIRI$-specific parent triples map graph
		$\triplesMapGraph{PTM}{\triplesMapIRI} \in \symRMLGraph$} is an RDF graph
	defined as follows:
	\begin{enumerate}
		\item  $\triplesMapGraph{PTM}{\triplesMapIRI}$  contains triples $(s,p,o) \in \symRMLGraph$
		      where $p = \texttt{rr:parentTriplesMap}$ and $o = \triplesMapIRI$.
		\item If $(s,\texttt{rr:parentTriplesMap},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains triples $(s',p',o') \in \symRMLGraph$
		      where $s'=s$ and  $p' = \texttt{rr:joinCondition}$.
		\item If $(s,\texttt{rr:joinCondition},o) \in \triplesMapGraph{PTM}{\triplesMapIRI}$, then
		      $\triplesMapGraph{PTM}{\triplesMapIRI}$ also contains triples $(s',p',o') \in \symRMLGraph$
		      where $s' = o$ and  $p' = \texttt{rr:parent}$.
	\end{enumerate}
\end{definition}
\begin{algorithm}[thbp]
	\caption{QueryInTriplesMap: Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
	\begin{algorithmic}[1]
		\Require  $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ the subject IRI of a triples map
		\Ensure  $\symDataAcc$ a set of query expressions
		\State $\symDataAcc \gets \emptyset$ \Comment Initial empty set of query expressions
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rml:reference}$ and  $o\in \symAllLiterals$}
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\For{$(s,p ,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ such that $p =\texttt{rr:template}$ and  $o\in \symAllLiterals$}
		    \For{$\queryExpr \in$ substrings in $o$ that are enclosed by \{ and \} }
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		    \EndFor
		\EndFor
        \State $T_{child} \gets \{ (s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI} \} $ where $p = \texttt{rr:child}$
        \State $T_{parent} \gets \{(s,p,o) \in \triplesMapGraph{PTM}{\triplesMapIRI}\}$ where $p = \texttt{rr:parent}$
        \For{$(s,p,o) \in T_{child} \cup T_{parent}$ such that $o \in \symAllLiterals$ }
		    \State $\queryExpr \gets $ lexical form of $o$
		    \State $\symDataAcc \gets \symDataAcc \cup \{ \queryExpr \}$
		\EndFor
		\State \Return $\symDataAcc$
	\end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{TranslateSourceOperator: Source operator translation for a given triples map IRI $\triplesMapIRI$}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the source operator for
		\Ensure $(\queryExpr_{root}, \symAttrQueryMap)$
        \State $ls_{iri} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = \triplesMapIRI$  and $p= \texttt{rml:logicalSource}$
        \State $\queryExpr_{root} \gets o$ for a triple $(s,p,o) \in \symRMLGraph$ where $s = ls_{iri}$  and $p=  \texttt{rml:iterator }$
        \State $\symDataAcc \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$} \Comment{Algorithm~\ref{alg:query_extraction}}
        \State $\symAttrQueryMap \gets \{ (a, \queryExpr) \ | \ \queryExpr \in \symDataAcc \text{ and } 
                                     a \in \symAttrUniverse  \text{ such that $a$ is unique in $\symAttrQueryMap$ }\}$  
		\State \Return  $(\queryExpr_{root}, \symAttrQueryMap)$ 
	\end{algorithmic}
\end{algorithm}

\begin{definition}
    A \textbf{normalized} RML document is defined by undertaking the following steps as described 
    in~\citeneeded
    \begin{enumerate}
        \item All shortcuts for constant-term maps have been expanded,
        \item All \texttt{rr:class} definitions are replaced by predicate object maps,
        \item All predicate-object maps with multiple predicate or object
            definitions are expanded into predicate-object maps with a single
            predicate and a single object definition,
        \item All triple maps with multiple predicate-object maps have been
            replaced by a set of equivalent triple maps.
    \end{enumerate}
    
\end{definition}


\begin{algorithm}
    \caption{RML translation to algebraic mapping expressions}
    \label{alg:rml2algebra}
    \begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of a \textbf{normalized} RML document
        \State $\omega \gets$  initialization value for the current algebraic mapping expressions
        \For{$(\triplesMapIRI,p,o) \in \symRMLGraph$ where  $p = \texttt{rdf:type}$ and $o= \texttt{rr:TriplesMap}$ }
        \State  $\iterConfig \gets \iterConfigTuple = $ \Call{TranslateSourceOperator}{$\symRMLGraph, \triplesMapIRI$} \Comment{Algorithm~\ref{alg:rml_source_op}}
        \State   $s \gets $ dataset represented by $o$ in $(\triplesMapIRI,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ where $p = \texttt{rr:logicalSource}$
        \State   $\omega \gets \fctSourceOp{\iterConfig}{s}$
        \State  $(v_{subj}, v_{pred},v_{obj}) \gets $ initial placeholder value for the (s,p,o) attributes for RDF serialization
            \For{$(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ where $p= \texttt{rr:subjectMap}$}
                \State $(a, \extExpr) \gets $ extend operator from triple $(o,\texttt{rr:template},o') \lor (o,\texttt{rr:reference},o')$ 
                \State $\extExpr \gets $\Call{TypeCast}{$\extExpr$} according to \texttt{rr:termType}
                \State  $v_{subj} \gets a$ 
                \State $\omega \gets \extOp(\omega)$
            \EndFor
            \For{$(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ where $p= \texttt{rr:predicateMap}$}
                \State $(a, \extExpr) \gets $ extend operator from triple $(o,\texttt{rr:template},o') \lor (o,\texttt{rr:reference},o')$ 
                \State $\extExpr \gets $\Call{TypeCast}{$\extExpr$} according to \texttt{rr:termType}
                \State  $v_{pred} \gets a$ 
                \State $\omega \gets \extOp(\omega)$
            \EndFor
            \For{$(s,p,o) \in \triplesMapGraph{TM}{\triplesMapIRI}$ where $p= \texttt{rr:objectMap}$}
                \State $(a, \extExpr) \gets $ extend operator from triple $(o,\texttt{rr:template},o') \lor (o,\texttt{rr:reference},o')$ 
                \State $\extExpr \gets $\Call{TypeCast}{$\extExpr$} according to \texttt{rr:termType}
                \State  $v_{obj} \gets a$ 
                \State $\omega \gets \extOp(\omega)$
            \EndFor
        \State $\omega \gets $ \Call{SerializeOp}{$(v_{subj}, v_{pred},v_{obj}, \omega)$}
        \State $Execute(\omega)$ 
        \State $\omega \gets$  initialization value for the current algebraic mapping expressions
        \EndFor
    \end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{Projection operator translation}
	\label{alg:projection_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document,
		$\triplesMapIRI$ subject IRI of the triples map to translate the projection operator for
		\Ensure $\symProjSet$ a set of projection attributes
		\Procedure{TranslateProjectionOperator}{$\triplesMapIRI,\symRMLGraph$}
		\State $\symAttrQueryMap \gets$ \Call{QueryInTriplesMap}{$\triplesMapIRI, \symRMLGraph$}
		\State $\symProjSet \gets \fctDom{\symAttrQueryMap}$
		\State \Return  $\symProjSet$
		\EndProcedure
	\end{algorithmic}
\end{algorithm}
