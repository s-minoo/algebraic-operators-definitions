\section{RML translation}%
\label{sec:rml_translation}


\begin{algorithm}[thbp]
	\caption{Extract query expressions from RML triples maps}
	\label{alg:query_extraction}
    \begin{algorithmic}[1]
        \Require  $\symRMLGraph$ an RDF graph of RML document, 
        $tm_{iri}$ the subject IRI of a triples map
        \Ensure  $Q_{pairs}$ a map of attributes to query expressions
        \Procedure{QueryInTriplesMap}{$tm_{iri}, \symRMLGraph$}
            \State $Q_{pairs} \gets$ hash map with type: $\symAttrUniverse \setminus \{\fragAttr\} \rightarrow \symDataAcc$ 
            \State $Q_{pairs} \gets$ \Call{QueryInTemplate}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State $Q_{pairs} \gets$ \Call{QueryInReferences}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State \Return $Q_{pairs}$
        \EndProcedure
        \Statex
        \Function{QueryInReferences}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
            \State $p_{ref} \gets$ rml:reference
            \For{$trip \in$ triples with predicate $p_{temp}$, associated with $tm_{iri}$ in $\symRMLGraph$}
                \State $\queryExpr \gets trip.o$\Comment{object literals of rml:reference are query expressions}
                \State $\attr \gets$ unique attribute derived based on $\queryExpr$
                \State $Q_{pairs}[\attr] \gets \queryExpr$
            \EndFor
            \State \Return $Q_{pairs}$
        \EndFunction

        \Statex
        \Function{QueryInTemplate}{$tm_{iri}, \symRMLGraph, Q_{pairs}$}
        \State $p_{temp} \gets$ rr:template
            \For{$trip \in$ triples with predicate $p_{temp}$, associated with $tm_{iri}$ in $\symRMLGraph$}
            \State $lit_{temp} \gets trip.o$\Comment{object literal with template string}
                \For{$\queryExpr \in$ query expressions surrounded by ``\{ \}" in $lit_{temp}$}
                    \State $\attr \gets$ unique attribute derived based on $\queryExpr$
                    \State $Q_{pairs}[\attr] \gets \queryExpr$
                \EndFor
            \EndFor
        \State \Return $Q_{pairs}$
        \EndFunction
    \end{algorithmic}
\end{algorithm}


\begin{algorithm}
	\caption{Source operator translation}
	\label{alg:rml_source_op}
	\begin{algorithmic}[1]
		\Require $\symRMLGraph$ an RDF graph of RML document
		\State $p \gets $ rml:logicalSource
		\State $tm_{iris}, ls_{iris} \gets $ subjects and objects in $\symRMLGraph$ with predicate $p$
        \\
		\For{$tm_{current} \gets tm_{iris} $}
        \State \Call{QueryInTemplate}{$tm_{iri}$,$\symRMLGraph$}
        \State  $p_{ref} \gets $  rml:reference
        \State  $tm_{current}$
		\EndFor
	\end{algorithmic}
\end{algorithm}
